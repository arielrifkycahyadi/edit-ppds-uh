                passwordInput.type =

                    ?

                    :
                    'Tampilkan password'
                );


                const icon =
                    toggleButton.querySelector(
                        '.material-symbols-outlined'
                    );


                if (icon) {


                    icon.textContent =
                        isHidden
                        ?
                        'visibility_off'
                        :
                        'visibility';


                }


            }
        );

        }


        alertBox.textContent =
            message;


        alertBox.className =
            'spj26auth-alert is-error';


    }


    function clearError() {


        if (!alertBox) {
            return;
        }


        alertBox.textContent =
            '';


        alertBox.className =
            'spj26auth-alert';


    }


    /* =====================================================
       SUBMIT STATE
    ===================================================== */

    function setSubmitting(
        isSubmitting
    ) {


        if (!submitButton) {
            return;
        }


        submitButton.disabled =
            isSubmitting;


        submitButton.style.opacity =
            isSubmitting
            ?
            '0.72'
            :
            '';


        submitButton.style.cursor =
            isSubmitting
            ?
            'wait'
            :
            'pointer';


        if (submitLabel) {


            submitLabel.textContent =
                isSubmitting
                ?
                'Memproses...'
                :
                'Masuk';


        }


    }


    /* =====================================================
       LOGIN FORM
       REAL WORDPRESS AUTHENTICATION
    ===================================================== */

    if (form) {


        form.addEventListener(
            'submit',
            async function (event) {


                event.preventDefault();


                clearError();


                const loginValue =
                    loginInput
                    ?
                    loginInput.value.trim()
                    :
                    '';


                const passwordValue =
                    passwordInput
                    ?
                    passwordInput.value
                    :
                    '';


                const rememberInput =
                    root.querySelector(
                        '#spj26auth-remember'
                    );


                /* =========================================
                   BASIC VALIDATION
                ========================================== */

                if (
                    !loginValue
                    ||
                    !passwordValue
                ) {


                    showError(
                        'Username/NIM dan password wajib diisi.'
                    );


                    return;


                }


                /* =========================================
                   CONFIG CHECK
                ========================================== */

                if (
                    !window.spj26AuthConfig
                    ||
                    !window.spj26AuthConfig.ajaxUrl
                    ||
                    !window.spj26AuthConfig.nonce
                ) {


                    showError(
                        'Konfigurasi autentikasi SIPATUJU belum tersedia. Pastikan snippet 000 — 01 aktif.'
                    );


                    return;


                }


                /* =========================================
                   REQUEST BODY
                ========================================== */

                const requestBody =
                    new URLSearchParams();


                requestBody.append(
                    'action',
                    'spj26_login_ajax'
                );


                requestBody.append(
                    'nonce',
                    window.spj26AuthConfig.nonce
                );


                requestBody.append(
                    'login_id',
                    loginValue
                );


                requestBody.append(
                    'password',
                    passwordValue
                );


                requestBody.append(
                    'remember',
                    rememberInput
                    &&
                    rememberInput.checked
                    ?
                    '1'
                    :
                    '0'
                );


                setSubmitting(
                    true
                );


                /* =========================================
                   AJAX LOGIN
                ========================================== */

                try {


                    const response =
                        await fetch(
                            window.spj26AuthConfig.ajaxUrl,
                            {

                                method:
                                    'POST',

                                credentials:
                                    'same-origin',

                                headers: {

                                    'Content-Type':
                                        'application/x-www-form-urlencoded; charset=UTF-8'

                                },

                                body:
                                    requestBody.toString()

                            }
                        );


                    let result =
                        null;


                    try {


                        result =
                            await response.json();


                    }
                    catch (jsonError) {


                        throw new Error(
                            'Respons server tidak valid.'
                        );


                    }


                    /* =====================================
                       LOGIN FAILURE
                    ====================================== */

                    if (
                        !result
                        ||
                        result.success
                        !==
                        true
                    ) {


                        const message =
                            result
                            &&
                            result.data
                            &&
                            result.data.message
                            ?
                            result.data.message
                            :
                            'Login gagal. Silakan periksa akun Anda.';


                        showError(
                            message
                        );


                        return;


                    }


                    /* =====================================
                       REDIRECT
                    ====================================== */

                    const redirectUrl =
                        result.data
                        &&
                        result.data.redirect
                        ?
                        result.data.redirect
                        :
                        '';


                    if (!redirectUrl) {


                        showError(
                            'Login berhasil, tetapi tujuan dashboard tidak ditemukan.'
                        );


                        return;


                    }


                    window.location.href =
                        redirectUrl;


                }
                catch (error) {


                    showError(
                        'Tidak dapat menghubungi sistem autentikasi. Silakan coba kembali.'
                    );


                }
                finally {


                    setSubmitting(
                        false
                    );


                }


            }
        );


    }


})();

</script>
