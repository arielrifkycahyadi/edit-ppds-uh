    async function backendRequest(
        action,
        payload
    ){

        const config =
            window.spj26ReviewerPortalConfig;

        if(
            !config
            ||
            !config.ajaxUrl
            ||
            !config.nonce
        ){

            throw new Error(
                'Konfigurasi backend Reviewer belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            config.nonce
        );


        Object.keys(
            payload || {}
        )
        .forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                config.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        let result = null;

        try{

            result =
                await response.json();

        }catch(error){

            throw new Error(
                'Respons server tidak valid.'
            );

        }


        if(
            !result
            ||
            result.success !== true
        ){

            throw new Error(
                result
                &&
                result.data
                &&
                result.data.message
                ?
                result.data.message
                :
                'Permintaan ke server gagal.'
            );

        }


        return result.data;

    }

    function finalResult(item){

        const journals =
            Array.isArray(
                item &&
                item.journals
            )
            ?
            item.journals
            :
            [];


        if(
            journals.some(function(journal){
                return (
                    journal.decision ===
                    'tidak_layak'
                );
            })
        ){

            return {
                key:'not_eligible',
                label:'Tidak Layak',
                className:'not-eligible'
            };

        }


        if(
            journals.length === 3
            &&
            journals.every(function(journal){
                return (
                    journal.decision ===
                    'layak'
                );
            })
        ){

            return {
                key:'eligible',
                label:'Layak',
                className:'eligible'
            };

        }


        return {
            key:'completed',
            label:'Selesai',
            className:'completed'
        };

    }


    function searchableText(item){

        const journals =
            Array.isArray(
                item.journals
            )
            ?
            item.journals
            :
            [];


        const journalText =
            journals
            .map(function(journal){

                return [
                    journal.name,
                    journal.issn,
                    journal.quartile,
                    journal.decision,
                    journal.comment
                ]
                .filter(Boolean)
                .join(' ');

            })
            .join(' ');


        return normalize(
            [
                item.code,
                item.resident_name,
                item.resident_nim,
                item.resident_program,
                item.article_title,
                item.article_type,
                item.supervisor_name,
                journalText
            ]
            .filter(Boolean)
            .join(' ')
        );

    }


    function getQuartileSummary(item){

        const journals =
            Array.isArray(
                item &&
                item.journals
            )
            ?
            item.journals
            :
            [];

        const unique = [];

        journals.forEach(function(journal){

            const value =
                journal &&
                journal.quartile
                ?
                String(
                    journal.quartile
                ).trim()
                :
                '';

            if(
                value
                &&
                !unique.includes(value)
            ){
                unique.push(value);
            }

        });

        return unique.join(' • ');

    }

    function getCounts(){

        const counts = {
            all:
                completedSubmissions.length,
            eligible:0,
            not_eligible:0
        };


        completedSubmissions
        .forEach(function(item){

            const status =
                finalResult(item);

            if(
                status.key === 'eligible'
            ){
                counts.eligible += 1;
            }

            if(
                status.key === 'not_eligible'
            ){
                counts.not_eligible += 1;
            }

        });


        return counts;

    }

    function getPageSize(){

        if(itemsPerPage === 'all'){

            return Math.max(
                filteredSubmissions.length,
                1
            );

        }

        return Math.max(
            1,
            Number(itemsPerPage) || 10
        );

    }


    function applyFilters(){

        const query =
            normalize(
                searchTerm
            );


        filteredSubmissions =
            completedSubmissions
            .filter(function(item){

                const status =
                    finalResult(item);


                if(
                    currentFilter !== 'all'
                    &&
                    status.key !== currentFilter
                ){
                    return false;
                }


                if(
                    query
                    &&
                    !searchableText(item)
                        .includes(query)
                ){
                    return false;
                }


                return true;

            });


        const pageSize =
            getPageSize();

        const totalPages =
            Math.max(
                1,
                Math.ceil(
                    filteredSubmissions.length /
                    pageSize
                )
            );


        if(currentPage > totalPages){
            currentPage = totalPages;
        }

        if(currentPage < 1){
            currentPage = 1;
        }


        renderTable();
        renderPagination();

    }

    async function loadHistory(
        showSuccess
    ){

        try{

            syncDebugConfig();

            let data = null;

            try{

                data =
                    await backendRequest(
                        'spj26_reviewer_dashboard_data',
                        {}
                    );

                debugState.dashboard =
                    'OK';

            }catch(dashboardError){

                debugState.dashboard =
                    'ERROR: ' +
                    (
                        dashboardError.message
                        ||
                        'unknown'
                    );

                debugState.lastError =
                    debugState.dashboard;

                renderDebug();

                throw dashboardError;

            }


            let completed =
                Array.isArray(
                    data &&
                    data.completed
                )
                ?
                data.completed
                :
                [];


            debugState.available =
                Array.isArray(
                    data &&
                    data.available
                )
                ?
                data.available.length
                :
                0;

            debugState.active =
                Array.isArray(
                    data &&
                    data.active
                )
                ?
                data.active.length
                :
                0;

            debugState.completed =
                completed.length;


            /*
             * Endpoint khusus hanya fallback jika completed Dashboard kosong.
             */
            if(
                completed.length === 0
            ){

                try{

                    const historyData =
                        await backendRequest(
                            'spj26_reviewer_history_data',
                            {}
                        );

                    debugState.history =
                        'OK';

                    if(
                        historyData
                        &&
                        Array.isArray(
                            historyData.completed
                        )
                    ){

                        if(
                            historyData.completed.length > 0
                        ){
                            completed =
                                historyData.completed;
                        }

                        debugState.completed =
                            completed.length;
                    }

                }catch(historyEndpointError){

                    debugState.history =
                        'ERROR: ' +
                        (
                            historyEndpointError.message
                            ||
                            'unknown'
                        );

                    debugState.lastError =
                        debugState.history;

                    console.warn(
                        'SIPATUJU Reviewer History: fallback endpoint khusus tidak tersedia.',
                        historyEndpointError
                    );

                }

            }else{

                debugState.history =
                    'tidak diperlukan';

            }


            completedSubmissions =
                completed;


            setText(
                'spj26rev-nav-available',
                debugState.available
            );

            setText(
                'spj26rev-nav-active',
                debugState.active
            );


            renderFilters();

            currentPage = 1;

            applyFilters();
            updateTime();

            debugState.lastError =
                '-';

            renderDebug();


            if(showSuccess){

                showToast(
                    'Riwayat pemeriksaan berhasil diperbarui.'
                );

            }


        }catch(error){

            console.error(
                'SIPATUJU Reviewer History:',
                error
            );

            debugState.lastError =
                'ERROR: ' +
                (
                    error.message
                    ||
                    'Riwayat pemeriksaan gagal dimuat.'
                );

            renderDebug();

            showToast(
                error.message ||
                'Riwayat pemeriksaan gagal dimuat.'
            );

        }

    }


    let historyInitialized = false;


    function hasReviewerConfig(){

        return !!(
            window.spj26ReviewerPortalConfig
            &&
            window.spj26ReviewerPortalConfig.ajaxUrl
            &&
            window.spj26ReviewerPortalConfig.nonce
        );

    }


    function initialize(){

        if(historyInitialized){
            return;
        }

        syncDebugConfig();

        if(!hasReviewerConfig()){

            debugState.lastError =
                'ERROR: Reviewer config belum tersedia untuk URL ini.';

            renderDebug();

            showToast(
                'Menunggu konfigurasi backend Reviewer...'
            );

            let attempts = 0;

            const waitTimer =
                window.setInterval(
                    function(){

                        attempts += 1;

                        if(hasReviewerConfig()){

                            window.clearInterval(
                                waitTimer
                            );

                            syncDebugConfig();
                            initialize();
                            return;
                        }

                        if(attempts >= 20){

                            window.clearInterval(
                                waitTimer
                            );

                            debugState.lastError =
                                'ERROR: Reviewer config tidak dicetak backend pada URL ini.';

                            renderDebug();

                            showToast(
                                'Konfigurasi backend Reviewer belum tersedia.'
                            );
                        }

                    },
                    250
                );

            return;
        }


        historyInitialized = true;

        renderIdentity();

        loadHistory(false);


        window.setInterval(
            function(){
                loadHistory(false);
            },
            30000
        );

    }


    if(
        document.readyState ===
        'complete'
    ){

        initialize();

    }else{

        window.addEventListener(
            'load',
            initialize,
            { once:true }
        );

        document.addEventListener(
            'DOMContentLoaded',
            initialize,
            { once:true }
        );

    }

})();

    function normalizePath(value){

        const links = root.querySelectorAll('.spj26rev-nav-link[href]');

        badge.className = 'spj26rev-nav-badge';
        badge.textContent = '0';

        const dot = document.createElement('span');
        dot.className = 'spj26rev-nav-dot';
        dot.hidden = true;

        wrap.appendChild(badge);
        wrap.appendChild(dot);
        link.appendChild(wrap);

        return {link:link,wrap:wrap,badge:badge,dot:dot};
    }

    const indicators = {
        available:ensureIndicator('available'),
        active:ensureIndicator('active'),
        completed:ensureIndicator('completed')
    };

    function getConfig(){
        return window.spj26ReviewerPortalConfig || null;
    }

    function getIdentity(){
        const cfg = getConfig() || {};
        return String(cfg.username || cfg.name || 'reviewer')
            .trim()
            .replace(/[^a-zA-Z0-9_-]/g,'_');
    }

    function getStorageKey(){
        return 'spj26rev_sidebar_seen_' + STORAGE_VERSION + '_' + getIdentity();
    }

    function buildSnapshot(items){
        const result = {};
        const list = Array.isArray(items) ? items : [];

        list.forEach(function(item){
            if(!item) return;

            const key = String(item.id || item.code || '').trim();
            if(!key) return;

            const journals = Array.isArray(item.journals)
                ? item.journals.map(function(journal){
                    return {
                        id:journal && journal.id ? String(journal.id) : '',
                        order:journal && journal.order ? String(journal.order) : '',
                        name:journal && journal.name ? String(journal.name) : '',
                        issn:journal && journal.issn ? String(journal.issn) : '',
                        quartile:journal && journal.quartile ? String(journal.quartile) : '',
                        link:journal && journal.link ? String(journal.link) : '',
                        decision:journal && journal.decision ? String(journal.decision) : '',
                        comment:journal && journal.comment ? String(journal.comment) : ''
                    };
                })
                : [];

            result[key] = JSON.stringify({
                code:item.code ? String(item.code) : '',
                status:item.status ? String(item.status) : '',
                status_label:item.status_label ? String(item.status_label) : '',
                assigned_at:item.assigned_at ? String(item.assigned_at) : '',
                updated_at:item.updated_at ? String(item.updated_at) : '',
                completed_at:item.completed_at ? String(item.completed_at) : '',
                journals:journals
            });
        });

        return result;
    }

    function diffCount(current, seen){
        const keys = new Set([
            ...Object.keys(current || {}),
            ...Object.keys(seen || {})
        ]);

        let count = 0;

        keys.forEach(function(key){
            if((current || {})[key] !== (seen || {})[key]){
                count += 1;
            }
        });

        return count;
    }

    function readState(){
        try{
            const raw = window.localStorage.getItem(getStorageKey());
            if(!raw) return null;

            const parsed = JSON.parse(raw);
            return parsed && typeof parsed === 'object'
                ? parsed
                : null;
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat dibaca.',
                error
            );
            return null;
        }
    }

    function writeState(state){
        try{
            window.localStorage.setItem(
                getStorageKey(),
                JSON.stringify(state)
            );
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat disimpan.',
                error
            );
        }
    }

    function setIndicator(section,total,unread){
        const indicator = indicators[section];
        if(!indicator) return;

        indicator.badge.textContent = String(total);
        indicator.dot.hidden = unread <= 0;

        indicator.wrap.title = unread > 0
            ? unread + ' pembaruan belum dilihat'
            : total + ' data saat ini';
    }

    async function requestData(){
        const cfg = getConfig();

        if(!cfg || !cfg.ajaxUrl || !cfg.nonce){
            throw new Error('Konfigurasi Reviewer belum tersedia.');
        }

        const body = new URLSearchParams();
        body.append('action','spj26_reviewer_dashboard_data');
        body.append('nonce',cfg.nonce);

        const response = await fetch(
            cfg.ajaxUrl,
            {
                method:'POST',
                credentials:'same-origin',
                headers:{
                    'Content-Type':'application/x-www-form-urlencoded; charset=UTF-8'
                },
                body:body.toString()
            }
        );

        const result = await response.json();

        if(!result || result.success !== true){
            throw new Error(
                result && result.data && result.data.message
                    ? result.data.message
                    : 'Data Reviewer gagal dimuat.'
            );
        }

        return result.data || {};
    }

    async function refreshSidebar(){
        try{
            const data = await requestData();

            const lists = {
                available:Array.isArray(data.available) ? data.available : [],
                active:Array.isArray(data.active) ? data.active : [],
                completed:Array.isArray(data.completed) ? data.completed : []
            };

            const current = {
                available:buildSnapshot(lists.available),
                active:buildSnapshot(lists.active),
                completed:buildSnapshot(lists.completed)
            };

            let state = readState();

            if(!state){
                state = {
                    available:current.available,
                    active:current.active,
                    completed:current.completed
                };
            }

            const unread = {
                available:diffCount(current.available,state.available || {}),
                active:diffCount(current.active,state.active || {}),
                completed:diffCount(current.completed,state.completed || {})
            };

            if(CURRENT_SECTION === 'available'){
                state.available = current.available;
                unread.available = 0;
            }

            if(CURRENT_SECTION === 'active'){
                state.active = current.active;
                unread.active = 0;
            }

            if(CURRENT_SECTION === 'completed'){
                state.completed = current.completed;
                unread.completed = 0;
            }

            setIndicator('available',lists.available.length,unread.available);
            setIndicator('active',lists.active.length,unread.active);
            setIndicator('completed',lists.completed.length,unread.completed);

            writeState(state);

            window.dispatchEvent(
                new CustomEvent(
                    'spj26:reviewer-sidebar-sync',
                    {detail:{data:data,unread:unread}}
                )
            );

        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: sinkronisasi notifikasi sidebar gagal.',
                error
            );
        }
    }

    function init(){
        refreshSidebar();
        window.setInterval(refreshSidebar,30000);
    }

    if(document.readyState === 'complete'){
        init();
    }else{
        window.addEventListener('load',init,{once:true});
    }
})();
</script>
