            sidebar.classList.add('open');
        }

        if(overlay){
            overlay.classList.add('open');
        }

    }


    function closeSidebar(){

        if(sidebar){
            sidebar.classList.remove('open');
        }

        if(overlay){
            overlay.classList.remove('open');
        }

    }

        menuButton.addEventListener(
            'click',
            openSidebar
        );
    }


    if(overlay){
        overlay.addEventListener(
            'click',
            closeSidebar
        );
    }

        .querySelectorAll(
            '.spj26rev-sidebar a'
        )
        .forEach(function(link){

            link.addEventListener(
                'click',
                function(){

                    if(
                        window.innerWidth <= 800
                    ){

            );

        });

        const element =
            root.querySelector(

                String(

    function showToast(message){

        if(!toast) return;

        toast.textContent =
            String(message || '');

        toast.hidden = false;


        window.setTimeout(
            function(){
                toast.hidden = true;
            },
            3200
        );

    }


    function createEl(
        tag,
        className,
        text
    ){

        const element =
            document.createElement(tag);

        if(className){
            element.className = className;
        }

        if(
            text !== undefined
            &&
            text !== null
        ){

            element.textContent = text;

        }

        return element;
    }


    /* =====================================================
       BACKEND
    ===================================================== */

    async function backendRequest(
        action,
        payload
    ){

        const config =
            window.spj26ReviewerPortalConfig;


        if(
            !config
            ||
            !config.ajaxUrl
            ||
            !config.nonce
        ){

            throw new Error(
                'Konfigurasi backend Reviewer belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            config.nonce
        );


        Object.keys(
            payload || {}
        )
        .forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                config.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result
            ||
            result.success !== true
        ){

            const message =
                result
                &&
                result.data
                &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

            throw new Error(message);

        }


        return result.data;

    }

        if(!config) return;


        setText(

        if(

                config.logoutUrl;

        }

    }

    function renderEmptyRow(
        title,
        message
    ){

        const tbody =
            root.querySelector(
                '#spj26rev-available-body'
            );

        if(!tbody) return;


        tbody.innerHTML = '';

            document.createElement('tr');

        const cell =
            document.createElement('td');

        cell.colSpan = 7;

            createEl(
                'div',
                'spj26rev-empty'
            );


        empty.appendChild(
            createEl(
                'span',
                'material-symbols-outlined',
                'inbox'
            )
        );

        empty.appendChild(
            createEl(
                'strong',
                '',
                title
            )
        );

        empty.appendChild(
            createEl(
                'span',
                '',
                message
            )
        );


        cell.appendChild(empty);
        row.appendChild(cell);
        tbody.appendChild(row);

    }

       TABLE
    ===================================================== */

    function buildResidentCell(item){

        const cell =
            document.createElement('td');


        cell.appendChild(
            createEl(

                'spj26rev-name',

        cell.appendChild(

        );


        return cell;
    }


    function buildArticleCell(item){

        const cell =
            document.createElement('td');


        cell.appendChild(
            createEl(
                'div',
                'spj26rev-name',
                item.article_title || '-'
            )
        );


        cell.appendChild(
            createEl(
                'div',
                'spj26rev-subtext',
                item.article_type || '-'
            )
        );


        return cell;
    }


    function renderTable(){

        const tbody =
            root.querySelector(
                '#spj26rev-available-body'
            );

        if(!tbody) return;


        setText(
            'spj26rev-available-count',
            filteredSubmissions.length
        );

        setText(
            'spj26rev-available-nav-count',
            availableSubmissions.length
        );


        if(
            availableSubmissions.length === 0
        ){

            renderEmptyRow(
                'Tidak ada pengajuan tersedia',
                'Pengajuan baru yang belum memiliki Reviewer akan muncul di halaman ini.'
            );

            return;

        }


        if(
            filteredSubmissions.length === 0
        ){

            renderEmptyRow(
                'Pengajuan tidak ditemukan',
                'Tidak ada pengajuan yang sesuai dengan kata kunci pencarian.'
            );

            return;

        }


        tbody.innerHTML = '';


        filteredSubmissions
            .forEach(function(item){

                const row =
                    document.createElement('tr');


                const codeCell =
                    document.createElement('td');

                codeCell.appendChild(
                    createEl(
                        'div',
                        'spj26rev-code',
                        item.code || '-'
                    )
                );


                const residentCell =
                    buildResidentCell(item);

                const articleCell =
                    buildArticleCell(item);


                const supervisorCell =
                    document.createElement('td');

                supervisorCell.appendChild(
                    createEl(
                        'div',
                        'spj26rev-name',
                        item.supervisor_name || '-'
                    )
                );


                const dateCell =
                    createEl(
                        'td',
                        '',
                        item.submitted_at || '-'
                    );


                const statusCell =
                    document.createElement('td');

                statusCell.appendChild(
                    createEl(
                        'span',
                        'spj26rev-status',
                        item.status_label ||
                        'Menunggu Pemeriksaan'
                    )
                );


                const actionCell =
                    document.createElement('td');

                const group =
                    createEl(
                        'div',
                        'spj26rev-action-group'
                    );


                const detailButton =
                    createEl(
                        'button',
                        'spj26rev-action',
                        'Detail'
                    );

                detailButton.type =
                    'button';

                detailButton.addEventListener(
                    'click',
                    function(){

                        openDetail(item);

                    }
                );


                const takeButton =
                    createEl(
                        'button',
                        'spj26rev-action primary',
                        'Ambil'
                    );

                takeButton.type =
                    'button';

                takeButton.addEventListener(
                    'click',
                    function(){

                        takeSubmission(
                            item.id,
                            takeButton
                        );

                    }
                );


                group.appendChild(
                    detailButton
                );

                group.appendChild(
                    takeButton
                );

                actionCell.appendChild(
                    group
                );


                row.appendChild(
                    codeCell
                );

                row.appendChild(
                    residentCell
                );

                row.appendChild(
                    articleCell
                );

                row.appendChild(
                    supervisorCell
                );

                row.appendChild(
                    dateCell
                );

                row.appendChild(
                    statusCell
                );

                row.appendChild(
                    actionCell
                );


                tbody.appendChild(
                    row
                );

            });

    }


    /* =====================================================
       SEARCH
    ===================================================== */

    const searchInput =
        root.querySelector(
            '#spj26rev-available-search'
        );


    function applySearch(){

        const query =
            searchInput
            ? searchInput.value
                .trim()
                .toLowerCase()
            : '';


        if(!query){

            filteredSubmissions =
                availableSubmissions.slice();

            renderTable();

            return;

        }


        filteredSubmissions =
            availableSubmissions
                .filter(function(item){

                    const haystack =
                        [
                            item.code,
                            item.resident_name,
                            item.resident_nim,
                            item.resident_program,
                            item.article_title,
                            item.article_type,
                            item.supervisor_name
                        ]
                        .filter(Boolean)
                        .join(' ')
                        .toLowerCase();


                    return haystack.includes(
                        query
                    );

                });


        renderTable();

    }


    if(searchInput){

        searchInput.addEventListener(
            'input',
            applySearch
        );

    }

    /* =====================================================
       TAKE SUBMISSION
    ===================================================== */

    async function takeSubmission(
        submissionId,
        button
    ){

        try{

            if(button){
                button.disabled = true;
            }


            await backendRequest(
                'spj26_reviewer_take_submission',
                {
                    submission_id:
                        submissionId
                }
            );


            closeDetail();


            showToast(
                'Pengajuan berhasil diambil. Pengajuan tersebut sekarang berada pada Pemeriksaan Saya.'
            );


            await loadAvailable();


        }catch(error){

            showToast(
                error.message ||
                'Pengajuan gagal diambil.'
            );


            await loadAvailable();


        }finally{

            if(button){
                button.disabled = false;
            }

        }

    }


    /* =====================================================
       LOAD DATA
    ===================================================== */

    async function loadAvailable(){

        try{

            const data =
                await backendRequest(
                    'spj26_reviewer_dashboard_data',
                    {}
                );


            availableSubmissions =
                Array.isArray(
                    data.available
                )
                ? data.available
                : [];


            filteredSubmissions =
                availableSubmissions.slice();


            if(searchInput){
                searchInput.value = '';
            }


            renderTable();


        }catch(error){

            console.error(
                'SIPATUJU Reviewer - Pengajuan Tersedia:',
                error
            );


            availableSubmissions = [];
            filteredSubmissions = [];


            renderTable();


            showToast(
                error.message ||
                'Data Pengajuan Tersedia gagal dimuat.'
            );

        }

    }


    const refreshButton =
        root.querySelector(
            '#spj26rev-available-refresh'
        );


    if(refreshButton){

        refreshButton.addEventListener(
            'click',
            loadAvailable
        );

    }


    /* =====================================================
       INITIAL
    ===================================================== */

    function initializeAvailablePage(){

        if(
            !window.spj26ReviewerPortalConfig
            ||
            !window.spj26ReviewerPortalConfig.ajaxUrl
            ||
            !window.spj26ReviewerPortalConfig.nonce
        ){

            console.error(
                'SIPATUJU Reviewer: spj26ReviewerPortalConfig tidak ditemukan.'
            );


            showToast(
                'Konfigurasi backend Reviewer belum tersedia.'
            );


            return;

        }


        renderIdentity();
        loadAvailable();

        window.setInterval(
            loadAvailable,
            30000
        );

    }


    if(
        document.readyState ===
        'complete'
    ){

        initializeAvailablePage();

    }else{

        window.addEventListener(
            'load',
            initializeAvailablePage,
            { once:true }
        );

    }

})();

<script>
(function(){
    const root = document.getElementById("spj26rev-available-root");
    if(!root) return;

    const CURRENT_SECTION = "available";
    const STORAGE_VERSION = 'v2';

    const PATHS = {
        available:'/reviewer-sipatuju/pengajuan-tersedia/',
        active:'/reviewer-sipatuju/pemeriksaan-saya/',
        completed:'/reviewer-sipatuju/riwayat-pemeriksaan/'
    };

    function normalizePath(value){
        try{
            return new URL(value, window.location.origin).pathname.replace(/\/+$/, '') + '/';
        }catch(error){
            return '';
        }
    }

    function findNavLink(path){
        const wanted = normalizePath(path);
        const links = root.querySelectorAll('.spj26rev-nav-link[href]');

        for(const link of links){
            if(normalizePath(link.getAttribute('href')) === wanted){
                return link;
            }
        }

        return null;
    }

    function ensureIndicator(section){
        const link = findNavLink(PATHS[section]);
        if(!link) return null;

        Array.from(link.children).forEach(function(node){
            if(
                node.classList
                &&
                (
                    node.classList.contains('spj26rev-nav-badge')
                    ||
                    node.classList.contains('spj26rev-nav-count')
                    ||
                    node.classList.contains('spj26rev-nav-notification')
                )
            ){
                node.remove();
            }
        });

        const wrap = document.createElement('span');
        wrap.className = 'spj26rev-nav-notification';

        const badge = document.createElement('span');
        badge.className = 'spj26rev-nav-badge';
        badge.textContent = '0';

        const dot = document.createElement('span');
        dot.className = 'spj26rev-nav-dot';
        dot.hidden = true;

        wrap.appendChild(badge);
        wrap.appendChild(dot);
        link.appendChild(wrap);

        return {link:link,wrap:wrap,badge:badge,dot:dot};
    }

    const indicators = {
        available:ensureIndicator('available'),
        active:ensureIndicator('active'),
        completed:ensureIndicator('completed')
    };

    function getConfig(){
        return window.spj26ReviewerPortalConfig || null;
    }

    function getIdentity(){
        const cfg = getConfig() || {};
        return String(cfg.username || cfg.name || 'reviewer')
            .trim()
            .replace(/[^a-zA-Z0-9_-]/g,'_');
    }

    function getStorageKey(){
        return 'spj26rev_sidebar_seen_' + STORAGE_VERSION + '_' + getIdentity();
    }

    function buildSnapshot(items){
        const result = {};
        const list = Array.isArray(items) ? items : [];

        list.forEach(function(item){
            if(!item) return;

            const key = String(item.id || item.code || '').trim();
            if(!key) return;

            const journals = Array.isArray(item.journals)
                ? item.journals.map(function(journal){
                    return {
                        id:journal && journal.id ? String(journal.id) : '',
                        order:journal && journal.order ? String(journal.order) : '',
                        name:journal && journal.name ? String(journal.name) : '',
                        issn:journal && journal.issn ? String(journal.issn) : '',
                        quartile:journal && journal.quartile ? String(journal.quartile) : '',
                        link:journal && journal.link ? String(journal.link) : '',
                        decision:journal && journal.decision ? String(journal.decision) : '',
                        comment:journal && journal.comment ? String(journal.comment) : ''
                    };
                })
                : [];

            result[key] = JSON.stringify({
                code:item.code ? String(item.code) : '',
                status:item.status ? String(item.status) : '',
                status_label:item.status_label ? String(item.status_label) : '',
                assigned_at:item.assigned_at ? String(item.assigned_at) : '',
                updated_at:item.updated_at ? String(item.updated_at) : '',
                completed_at:item.completed_at ? String(item.completed_at) : '',
                journals:journals
            });
        });

        return result;
    }

    function diffCount(current, seen){
        const keys = new Set([
            ...Object.keys(current || {}),
            ...Object.keys(seen || {})
        ]);

        let count = 0;

        keys.forEach(function(key){
            if((current || {})[key] !== (seen || {})[key]){
                count += 1;
            }
        });

        return count;
    }

    function readState(){
        try{
            const raw = window.localStorage.getItem(getStorageKey());
            if(!raw) return null;

            const parsed = JSON.parse(raw);
            return parsed && typeof parsed === 'object'
                ? parsed
                : null;
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat dibaca.',
                error
            );
            return null;
        }
    }

    function writeState(state){
        try{
            window.localStorage.setItem(
                getStorageKey(),
                JSON.stringify(state)
            );
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat disimpan.',
                error
            );
        }
    }

    function setIndicator(section,total,unread){
        const indicator = indicators[section];
        if(!indicator) return;

        indicator.badge.textContent = String(total);
        indicator.dot.hidden = unread <= 0;

        indicator.wrap.title = unread > 0
            ? unread + ' pembaruan belum dilihat'
            : total + ' data saat ini';
    }

    async function requestData(){
        const cfg = getConfig();

        if(!cfg || !cfg.ajaxUrl || !cfg.nonce){
            throw new Error('Konfigurasi Reviewer belum tersedia.');
        }

        const body = new URLSearchParams();
        body.append('action','spj26_reviewer_dashboard_data');
        body.append('nonce',cfg.nonce);

        const response = await fetch(
            cfg.ajaxUrl,
            {
                method:'POST',
                credentials:'same-origin',
                headers:{
                    'Content-Type':'application/x-www-form-urlencoded; charset=UTF-8'
                },
                body:body.toString()
            }
        );

        const result = await response.json();

        if(!result || result.success !== true){
            throw new Error(
                result && result.data && result.data.message
                    ? result.data.message
                    : 'Data Reviewer gagal dimuat.'
            );
        }

        return result.data || {};
    }

    async function refreshSidebar(){
        try{
            const data = await requestData();

            const lists = {
                available:Array.isArray(data.available) ? data.available : [],
                active:Array.isArray(data.active) ? data.active : [],
                completed:Array.isArray(data.completed) ? data.completed : []
            };

            const current = {
                available:buildSnapshot(lists.available),
                active:buildSnapshot(lists.active),
                completed:buildSnapshot(lists.completed)
            };

            let state = readState();

            if(!state){
                state = {
                    available:current.available,
                    active:current.active,
                    completed:current.completed
                };
            }

            const unread = {
                available:diffCount(current.available,state.available || {}),
                active:diffCount(current.active,state.active || {}),
                completed:diffCount(current.completed,state.completed || {})
            };

            if(CURRENT_SECTION === 'available'){
                state.available = current.available;
                unread.available = 0;
            }

            if(CURRENT_SECTION === 'active'){
                state.active = current.active;
                unread.active = 0;
            }

            if(CURRENT_SECTION === 'completed'){
                state.completed = current.completed;
                unread.completed = 0;
            }

            setIndicator('available',lists.available.length,unread.available);
            setIndicator('active',lists.active.length,unread.active);
            setIndicator('completed',lists.completed.length,unread.completed);

            writeState(state);

            window.dispatchEvent(
                new CustomEvent(
                    'spj26:reviewer-sidebar-sync',
                    {detail:{data:data,unread:unread}}
                )
            );

        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: sinkronisasi notifikasi sidebar gagal.',
                error
            );
        }
    }

    function init(){
        refreshSidebar();
        window.setInterval(refreshSidebar,30000);
    }

    if(document.readyState === 'complete'){
        init();
    }else{
        window.addEventListener('load',init,{once:true});
    }
})();
</script>
