    async function backendRequest(
        action,
        payload
    ){

        const config =
            window.spj26ReviewerPortalConfig;


        if(
            !config
            ||
            !config.ajaxUrl
            ||
            !config.nonce
        ){

            throw new Error(
                'Konfigurasi backend Reviewer belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();


        body.append(
            'action',
            action
        );


        body.append(
            'nonce',
            config.nonce
        );


        Object.keys(
            payload || {}
        )
        .forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                config.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result
            ||
            result.success !== true
        ){

            const message =
                result
                &&
                result.data
                &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

                message
            );

        return result.data;

    }

    function safeJournalUrl(value){

        const raw =
            String(
                value || ''
            ).trim();

        if(!raw){
            return '';
        }

        try{

            const url =
                new URL(raw);

            if(
                url.protocol !== 'http:'
                &&
                url.protocol !== 'https:'
            ){
                return '';
            }

            return url.href;

        }catch(error){

            return '';
        }

    }



    const REJECTION_REASON_VALUES = {
        discontinued_predatory:'Discontinued / Predatory',
        scope_mismatch:'Scope Tidak Sesuai'
    };

    function parseStoredReviewComment(value){
        const raw = String(value || '').trim();
        const reasons = [];

        if(raw.includes(REJECTION_REASON_VALUES.discontinued_predatory)){
            reasons.push('discontinued_predatory');
        }

        if(raw.includes(REJECTION_REASON_VALUES.scope_mismatch)){
            reasons.push('scope_mismatch');
        }

        let note = raw;

        if(/^Alasan Tidak Layak:/i.test(note)){
            const lines = note.split(/\r?\n/);
            lines.shift();
            note = lines.join('\n').replace(/^Catatan Tambahan:\s*/i,'').trim();
        }

        return { reasons:reasons, note:note };
    }

    function buildStoredReviewComment(reasons,note){
        const labels = (reasons || [])
            .map(function(reason){ return REJECTION_REASON_VALUES[reason] || ''; })
            .filter(Boolean);

        let value = 'Alasan Tidak Layak: ' + labels.join('; ');
        const noteValue = String(note || '').trim();

        if(noteValue){
            value += '\nCatatan Tambahan: ' + noteValue;
        }

        return value;
    }

    async function loadSubmission(){

        clearError();


        const params =
            new URLSearchParams(
                window.location.search
            );


        const submissionId =
            Number(
                params.get(
                    'submission_id'
                )
            );


        if(
            !submissionId
            ||
            !Number.isInteger(
                submissionId
            )
            ||
            submissionId <= 0
        ){

            showError(
                'ID pengajuan tidak ditemukan. Buka pemeriksaan melalui halaman Pemeriksaan Saya.'
            );


            if(submitButton){

                submitButton.disabled =
                    true;

            }


            return;

        }


        try{

            const data =
                await backendRequest(
                    'spj26_reviewer_dashboard_data',
                    {}
                );


            const active =
                Array.isArray(
                    data.active
                )
                ? data.active
                : [];


            const item =
                active.find(
                    function(row){

                        return Number(
                            row.id
                        ) ===
                        submissionId;

                    }
                );


            if(!item){

                showError(
                    'Pengajuan tidak ditemukan pada Pemeriksaan Saya atau sudah selesai diperiksa.'
                );


                if(submitButton){

                    submitButton.disabled =
                        true;

                }


                return;

            }


            renderSubmission(
                item
            );


        }catch(error){

            console.error(
                'SIPATUJU Reviewer - Pemeriksaan Jurnal:',
                error
            );


            showError(
                error.message ||
                'Data pemeriksaan gagal dimuat.'
            );


            if(submitButton){

                submitButton.disabled =
                    true;

            }

        }

    }

    async function submitReview(){

        if(
            !currentSubmission
        ){

            showError(
                'Data pengajuan belum tersedia.'
            );


            return;

        }


        clearError();


        const journals =
            [];


        for(
            const journal of
            (
                currentSubmission.journals ||
                []
            )
        ){

            const decision =
                root.querySelector(
                    'input[name="spj26rev-decision-' +
                    journal.id +
                    '"]:checked'
                );


            const comment =
                root.querySelector(
                    '[data-review-comment="' +
                    journal.id +
                    '"]'
                );


            if(!decision){

                showError(
                    'Pilih keputusan Layak atau Tidak Layak untuk seluruh jurnal.'
                );


                return;

            }


            const commentValue =
                comment
                ? comment.value.trim()
                : '';


            let storedComment =
                commentValue;


            if(
                decision.value ===
                'tidak_layak'
            ){

                const reasonInputs =
                    root.querySelectorAll(
                        'input[data-review-reason="' +
                        journal.id +
                        '"]:checked'
                    );

                const reasons =
                    Array.from(
                        reasonInputs
                    )
                    .map(function(input){
                        return input.value;
                    });

                const reasonError =
                    root.querySelector(
                        '[data-rejection-error="' +
                        journal.id +
                        '"]'
                    );

                if(
                    reasons.length === 0
                ){

                    if(reasonError){
                        reasonError.textContent =
                            'Pilih minimal satu alasan Tidak Layak.';

                        reasonError.hidden =
                            false;

                        reasonError.scrollIntoView({
                            behavior:'smooth',
                            block:'center'
                        });
                    }

                    showError(
                        'Pilih minimal satu alasan untuk setiap jurnal yang dinyatakan Tidak Layak.'
                    );

                    return;
                }

                if(reasonError){
                    reasonError.hidden = true;
                    reasonError.textContent = '';
                }

                storedComment =
                    buildStoredReviewComment(
                        reasons,
                        commentValue
                    );
            }


            journals.push({
                id:
                    journal.id,
                decision:
                    decision.value,
                comment:
                    storedComment
            });

        }


        if(
            journals.length !== 3
        ){

            showError(
                'Pemeriksaan harus mencakup tepat tiga jurnal kandidat.'
            );


            return;

        }


        if(submitButton){

            submitButton.disabled =
                true;

        }


        try{

            await backendRequest(
                'spj26_reviewer_submit_review',
                {
                    submission_id:
                        currentSubmission.id,
                    journals:
                        JSON.stringify(
                            journals
                        )
                }
            );


            showToast(
                'Hasil pemeriksaan berhasil dikirim.'
            );


            window.setTimeout(
                function(){

                    window.location.href =
                        '/reviewer-sipatuju/riwayat-pemeriksaan/';

                },
                650
            );


        }catch(error){

            showError(
                error.message ||
                'Hasil pemeriksaan gagal dikirim.'
            );


            if(submitButton){

                submitButton.disabled =
                    false;

            }

        }

    }


    if(submitButton){

        submitButton.addEventListener(
            'click',
            submitReview
        );

    }

    if(
        document.readyState ===
        'complete'

        initializeReviewPage();

    }else{

        window.addEventListener(

<script>
(function(){
    const root = document.getElementById("spj26rev-review-root");
    if(!root) return;

    const CURRENT_SECTION = "active";
    const STORAGE_VERSION = 'v2';

    const PATHS = {
        available:'/reviewer-sipatuju/pengajuan-tersedia/',
        active:'/reviewer-sipatuju/pemeriksaan-saya/',
        completed:'/reviewer-sipatuju/riwayat-pemeriksaan/'
    };

    function normalizePath(value){
        try{
            return new URL(value, window.location.origin).pathname.replace(/\/+$/, '') + '/';
        }catch(error){
            return '';
        }
    }

    function findNavLink(path){
        const wanted = normalizePath(path);
        const links = root.querySelectorAll('.spj26rev-nav-link[href]');

        for(const link of links){
            if(normalizePath(link.getAttribute('href')) === wanted){
                return link;
            }
        }

        return null;
    }

    function ensureIndicator(section){
        const link = findNavLink(PATHS[section]);
        if(!link) return null;

        Array.from(link.children).forEach(function(node){
            if(
                node.classList
                &&
                (
                    node.classList.contains('spj26rev-nav-badge')
                    ||
                    node.classList.contains('spj26rev-nav-count')
                    ||
                    node.classList.contains('spj26rev-nav-notification')
                )
            ){
                node.remove();
            }
        });

        const wrap = document.createElement('span');
        wrap.className = 'spj26rev-nav-notification';

        const badge = document.createElement('span');
        badge.className = 'spj26rev-nav-badge';
        badge.textContent = '0';

        const dot = document.createElement('span');
        dot.className = 'spj26rev-nav-dot';
        dot.hidden = true;

        wrap.appendChild(badge);
        wrap.appendChild(dot);
        link.appendChild(wrap);

        return {link:link,wrap:wrap,badge:badge,dot:dot};
    }

    const indicators = {
        available:ensureIndicator('available'),
        active:ensureIndicator('active'),
        completed:ensureIndicator('completed')
    };

    function getConfig(){
        return window.spj26ReviewerPortalConfig || null;
    }

    function getIdentity(){
        const cfg = getConfig() || {};
        return String(cfg.username || cfg.name || 'reviewer')
            .trim()
            .replace(/[^a-zA-Z0-9_-]/g,'_');
    }

    function getStorageKey(){
        return 'spj26rev_sidebar_seen_' + STORAGE_VERSION + '_' + getIdentity();
    }

    function buildSnapshot(items){
        const result = {};
        const list = Array.isArray(items) ? items : [];

        list.forEach(function(item){
            if(!item) return;

            const key = String(item.id || item.code || '').trim();
            if(!key) return;

            const journals = Array.isArray(item.journals)
                ? item.journals.map(function(journal){
                    return {
                        id:journal && journal.id ? String(journal.id) : '',
                        order:journal && journal.order ? String(journal.order) : '',
                        name:journal && journal.name ? String(journal.name) : '',
                        issn:journal && journal.issn ? String(journal.issn) : '',
                        quartile:journal && journal.quartile ? String(journal.quartile) : '',
                        link:journal && journal.link ? String(journal.link) : '',
                        decision:journal && journal.decision ? String(journal.decision) : '',
                        comment:journal && journal.comment ? String(journal.comment) : ''
                    };
                })
                : [];

            result[key] = JSON.stringify({
                code:item.code ? String(item.code) : '',
                status:item.status ? String(item.status) : '',
                status_label:item.status_label ? String(item.status_label) : '',
                assigned_at:item.assigned_at ? String(item.assigned_at) : '',
                updated_at:item.updated_at ? String(item.updated_at) : '',
                completed_at:item.completed_at ? String(item.completed_at) : '',
                journals:journals
            });
        });

        return result;
    }

    function diffCount(current, seen){
        const keys = new Set([
            ...Object.keys(current || {}),
            ...Object.keys(seen || {})
        ]);

        let count = 0;

        keys.forEach(function(key){
            if((current || {})[key] !== (seen || {})[key]){
                count += 1;
            }
        });

        return count;
    }

    function readState(){
        try{
            const raw = window.localStorage.getItem(getStorageKey());
            if(!raw) return null;

            const parsed = JSON.parse(raw);
            return parsed && typeof parsed === 'object'
                ? parsed
                : null;
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat dibaca.',
                error
            );
            return null;
        }
    }

    function writeState(state){
        try{
            window.localStorage.setItem(
                getStorageKey(),
                JSON.stringify(state)
            );
        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: state notifikasi tidak dapat disimpan.',
                error
            );
        }
    }

    function setIndicator(section,total,unread){
        const indicator = indicators[section];
        if(!indicator) return;

        indicator.badge.textContent = String(total);
        indicator.dot.hidden = unread <= 0;

        indicator.wrap.title = unread > 0
            ? unread + ' pembaruan belum dilihat'
            : total + ' data saat ini';
    }

    async function requestData(){
        const cfg = getConfig();

        if(!cfg || !cfg.ajaxUrl || !cfg.nonce){
            throw new Error('Konfigurasi Reviewer belum tersedia.');
        }

        const body = new URLSearchParams();
        body.append('action','spj26_reviewer_dashboard_data');
        body.append('nonce',cfg.nonce);

        const response = await fetch(
            cfg.ajaxUrl,
            {
                method:'POST',
                credentials:'same-origin',
                headers:{
                    'Content-Type':'application/x-www-form-urlencoded; charset=UTF-8'
                },
                body:body.toString()
            }
        );

        const result = await response.json();

        if(!result || result.success !== true){
            throw new Error(
                result && result.data && result.data.message
                    ? result.data.message
                    : 'Data Reviewer gagal dimuat.'
            );
        }

        return result.data || {};
    }

    async function refreshSidebar(){
        try{
            const data = await requestData();

            const lists = {
                available:Array.isArray(data.available) ? data.available : [],
                active:Array.isArray(data.active) ? data.active : [],
                completed:Array.isArray(data.completed) ? data.completed : []
            };

            const current = {
                available:buildSnapshot(lists.available),
                active:buildSnapshot(lists.active),
                completed:buildSnapshot(lists.completed)
            };

            let state = readState();

            if(!state){
                state = {
                    available:current.available,
                    active:current.active,
                    completed:current.completed
                };
            }

            const unread = {
                available:diffCount(current.available,state.available || {}),
                active:diffCount(current.active,state.active || {}),
                completed:diffCount(current.completed,state.completed || {})
            };

            if(CURRENT_SECTION === 'available'){
                state.available = current.available;
                unread.available = 0;
            }

            if(CURRENT_SECTION === 'active'){
                state.active = current.active;
                unread.active = 0;
            }

            if(CURRENT_SECTION === 'completed'){
                state.completed = current.completed;
                unread.completed = 0;
            }

            setIndicator('available',lists.available.length,unread.available);
            setIndicator('active',lists.active.length,unread.active);
            setIndicator('completed',lists.completed.length,unread.completed);

            writeState(state);

            window.dispatchEvent(
                new CustomEvent(
                    'spj26:reviewer-sidebar-sync',
                    {detail:{data:data,unread:unread}}
                )
            );

        }catch(error){
            console.warn(
                'SIPATUJU Reviewer: sinkronisasi notifikasi sidebar gagal.',
                error
            );
        }
    }

    function init(){
        refreshSidebar();
        window.setInterval(refreshSidebar,30000);
    }

    if(document.readyState === 'complete'){
        init();
    }else{
        window.addEventListener('load',init,{once:true});
    }
})();
</script>
