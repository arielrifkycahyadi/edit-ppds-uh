    (function setupPagination(){

        const body = root.querySelector('#spj26adm-journal-table-body');
        const sizeSelect = root.querySelector('#spj26adm-journal-page-size');
        const info = root.querySelector('#spj26adm-journal-page-info');
        const prev = root.querySelector('#spj26adm-journal-page-prev');
        const next = root.querySelector('#spj26adm-journal-page-next');

        if(!body || !sizeSelect || !info || !prev || !next){
            return;
        }

        let page = 1;

        function getDataRows(){
            return Array.from(body.children).filter(function(row){
                return !row.querySelector('.spj26adm-empty-state');
            });
        }

        function applyPagination(){

            const rows = getDataRows();
            const total = rows.length;
            const all = sizeSelect.value === 'all';
            const requested = Number(sizeSelect.value) || 10;
            const pageSize = all ? Math.max(total,1) : requested;
            const totalPages = total === 0 ? 1 : Math.max(1, Math.ceil(total / pageSize));

            page = Math.min(Math.max(page,1), totalPages);

            const startIndex = all ? 0 : (page - 1) * pageSize;
            const endIndex = all ? total : Math.min(startIndex + pageSize,total);

            rows.forEach(function(row,index){
                row.hidden = !(index >= startIndex && index < endIndex);
            });

            if(total === 0){
                info.textContent = '0 data';
            }else if(all){
                info.textContent = total + ' data';
            }else{
                info.textContent = (startIndex + 1) + '–' + endIndex + ' dari ' + total + ' data';
            }

            prev.disabled = all || total === 0 || page <= 1;
            next.disabled = all || total === 0 || page >= totalPages;
        }

        sizeSelect.addEventListener('change',function(){
            page = 1;
            applyPagination();
        });

        prev.addEventListener('click',function(){
            if(page > 1){
                page -= 1;
                applyPagination();
            }
        });

        next.addEventListener('click',function(){
            page += 1;
            applyPagination();
        });

        [
            root.querySelector('#spj26adm-journal-search')
        ].filter(Boolean).forEach(function(el){
            el.addEventListener(
                el.tagName === 'INPUT' ? 'input' : 'change',
                function(){ page = 1; },
                { capture:true }
            );
        });

        if(window.MutationObserver){
            new MutationObserver(function(){
                applyPagination();
            }).observe(body,{ childList:true });
        }

        applyPagination();

    })();

            function(event){

                if(event.target === modal){
                    closeRemoveModal();
                }

            }
        );
    }


    /* =====================================================
       BACKEND REQUEST
       Endpoint/action ini disiapkan untuk integrasi backend:
       - spj26_admin_restricted_journals_list
       - spj26_admin_restricted_journal_remove
    ===================================================== */

    function getConfig(){

        return (
            window.spj26AdminJournalConfig ||
            window.spj26AdminDashboardConfig ||
            null
        );

    }


    async function backendRequest(
        action,
        payload
    ){

        const config =
            getConfig();


        if(
            !config ||
            !config.ajaxUrl ||
            !config.nonce
        ){

            throw new Error(
                'Konfigurasi backend halaman jurnal belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            config.nonce
        );


        Object.keys(
            payload || {}
        ).forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                config.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result ||
            result.success !== true
        ){

            const message =
                result &&
                result.data &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

            throw new Error(message);

        }


        return result.data;

    }


    async function loadJournals(){

        renderJournals();

        const config =
            getConfig();

        /*
         * Tidak ada dummy data.
         * Jika konfigurasi backend belum dipasang,
         * halaman tetap menampilkan keadaan kosong dengan aman.
         */
        if(
            !config ||
            !config.ajaxUrl ||
            !config.nonce
        ){
            return;
        }


        try{

            const data =
                await backendRequest(
                    'spj26_admin_restricted_journals_list',
                    {}
                );


            const journalCandidates = [
                data,
                data && data.journals,
                data && data.items,
                data && data.data,
                data && data.data && data.data.journals
            ];

            journalData =
                journalCandidates.find(Array.isArray) || [];


            showAlert('');
            renderJournals();


        }catch(error){

            console.error(
                'SIPATUJU Jurnal Predator / Discontinued:',
                error
            );

            showAlert(
                error.message ||
                'Daftar jurnal gagal dimuat dari backend SIPATUJU.'
            );

        }

    }


    if(confirmRemove){

        confirmRemove.addEventListener(
            'click',
            async function(){

                if(!activeJournalId){
                    return;
                }


                confirmRemove.disabled =
                    true;


                try{

                    await backendRequest(
                        'spj26_admin_restricted_journal_remove',
                        {
                            journal_id:
                                activeJournalId
                        }
                    );


                    const removedName =
                        activeJournalName;


                    closeRemoveModal();


                    showToast(
                        removedName
                        ? '"' + removedName + '" berhasil dikeluarkan dari daftar.'
                        : 'Jurnal berhasil dikeluarkan dari daftar.'
                    );


                    await loadJournals();


                }catch(error){

                    showToast(
                        error.message ||
                        'Jurnal gagal dikeluarkan dari daftar.'
                    );

                }finally{

                    confirmRemove.disabled =
                        false;

                }

            }
        );

    }


    /* =====================================================
       CONFIG / LOGOUT
    ===================================================== */

    function applyConfig(){

        const config =
            getConfig();

        const logoutLink =
            root.querySelector(
                '#spj26adm-logout'
            );


        if(
            logoutLink &&
            config &&
            config.logoutUrl
        ){

            logoutLink.href =
                config.logoutUrl;

        }

    }


    /* =====================================================
       INITIAL
    ===================================================== */

    function initializePage(){

        applyConfig();
        loadJournals();

    }


    if(
        document.readyState === 'complete'
    ){

        initializePage();

    }else{

        window.addEventListener(
            'load',
            initializePage,
            { once:true }
        );

    }

})();
</script>
