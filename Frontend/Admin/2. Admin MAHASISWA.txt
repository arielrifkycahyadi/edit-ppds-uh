(function(){

    /*
     * Desktop layout lock:
     * browser/page utama tidak scroll.
     * Scroll hanya terjadi pada area konten kanan.
     */
    document.documentElement.classList.add(
        'spj26adm-page-lock'
    );

    document.body.classList.add(
        'spj26adm-page-lock'
    );


    const root =
        document.getElementById(
            'spj26adm-residen-root'
        );

    if(!root) return;


    /* =====================================================
       MOBILE SIDEBAR
    ===================================================== */

    const menuButton =
        root.querySelector(
            '#spj26adm-residen-menu-button'
        );

    const sidebar =
        root.querySelector(
            '#spj26adm-residen-sidebar'
        );

    const overlay =
        root.querySelector(
            '#spj26adm-residen-overlay'
        );


    function openSidebar(){

        if(sidebar){
            sidebar.classList.add('open');
        }

        if(overlay){
            overlay.classList.add('open');
        }

    }


    function closeSidebar(){

        if(sidebar){
            sidebar.classList.remove('open');
        }

        if(overlay){
            overlay.classList.remove('open');
        }

    }


    if(menuButton){
        menuButton.addEventListener(
            'click',
            openSidebar
        );
    }

    if(overlay){
        overlay.addEventListener(
            'click',
            closeSidebar
        );
    }


    /* =====================================================
       PASSWORD VISIBILITY
    ===================================================== */

    root
        .querySelectorAll(
            '[data-password-toggle]'
        )
        .forEach(function(button){

            button.addEventListener(
                'click',
                function(){

                    const targetId =
                        button.getAttribute(
                            'data-password-toggle'
                        );

                    const input =
                        root.querySelector(
                            '#' + targetId
                        );

                    if(!input) return;


                    const showing =
                        input.type === 'text';


                    input.type =
                        showing
                        ? 'password'
                        : 'text';


                    const icon =
                        button.querySelector(
                            '.material-symbols-outlined'
                        );


                    if(icon){

                        icon.textContent =
                            showing
                            ? 'visibility'
                            : 'visibility_off';

                    }


                    button.setAttribute(
                        'aria-pressed',
                        showing
                        ? 'false'
                        : 'true'
                    );


                    button.setAttribute(
                        'aria-label',
                        showing
                        ? 'Lihat password'
                        : 'Sembunyikan password'
                    );

                }
            );

        });


    /* =====================================================
       STATE
    ===================================================== */

    async function backendRequest(
        action,
        payload
    ){

        if(
            !window.spj26ResidentAdminConfig ||
            !window.spj26ResidentAdminConfig.ajaxUrl ||
            !window.spj26ResidentAdminConfig.nonce
        ){

            throw new Error(
                'Konfigurasi backend Manajemen Mahasiswa belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            window.spj26ResidentAdminConfig.nonce
        );


        Object.keys(
            payload || {}
        ).forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                window.spj26ResidentAdminConfig.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result ||
            result.success !== true
        ){

            const message =
                result &&
                result.data &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

            throw new Error(message);

        }


        return result.data;

    }

    function renderMetrics(){

        const total =
            residents.length;

        const active =
            residents.filter(
                item => item.active === true
            ).length;

        const never =
            residents.filter(
                item =>
                item.active === true &&
                !item.last_login
            ).length;

        const inactive =
            residents.filter(
                item => item.active === false
            ).length;


        setText(
            'spj26adm-residen-count-total',
            total
        );

        setText(
            'spj26adm-residen-count-active',
            active
        );

        setText(
            'spj26adm-residen-count-never',
            never
        );

        setText(
            'spj26adm-residen-count-inactive',
            inactive
        );

    }

    function canonicalResidentProgram(value){

        const program =
            String(
                value || ''
            ).trim();

        /*
         * Backward compatibility:
         * akun lama dapat masih menyimpan ejaan "Orthopedi".
         * Di UI dan filter dinormalisasi menjadi nomenklatur baru.
         */
        if(
            program.toLowerCase() ===
            'orthopedi dan traumatologi'
        ){
            return 'Orthopaedi dan Traumatologi';
        }

        return program;
    }


    function getFilteredResidents(){

        const query =
            searchInput
            ? searchInput.value.trim().toLowerCase()
            : '';

        const program =
            programFilter
            ? programFilter.value
            : 'all';

        const status =
            statusFilter
            ? statusFilter.value
            : 'all';


        return residents.filter(function(item){

            const text =
                [
                    item.nim,
                    item.name,
                    item.program
                ]
                .filter(Boolean)
                .join(' ')
                .toLowerCase();


            const queryMatch =
                !query ||
                text.includes(query);


            const programMatch =
                program === 'all' ||
                item.program === program;


            let statusMatch = true;


            if(status === 'active'){

                statusMatch =
                    item.active === true;

            }else if(status === 'inactive'){

                statusMatch =
                    item.active === false;

            }else if(status === 'never'){

                statusMatch =
                    item.active === true &&
                    !item.last_login;

            }


            return (
                queryMatch &&
                programMatch &&
                statusMatch
            );

        });

    }

    /* =====================================================
       CLIENT-SIDE PAGINATION
       10 / 20 / 30 / Semua
    ===================================================== */
    (function setupPagination(){

        const body = root.querySelector('#spj26adm-residen-table-body');
        const sizeSelect = root.querySelector('#spj26adm-residen-page-size');
        const info = root.querySelector('#spj26adm-residen-page-info');
        const prev = root.querySelector('#spj26adm-residen-page-prev');
        const next = root.querySelector('#spj26adm-residen-page-next');

        if(!body || !sizeSelect || !info || !prev || !next){
            return;
        }

        let page = 1;

        function getDataRows(){
            return Array.from(body.children).filter(function(row){
                return !row.querySelector('.spj26adm-empty-state');
            });
        }

        function applyPagination(){

            const rows = getDataRows();
            const total = rows.length;
            const all = sizeSelect.value === 'all';
            const requested = Number(sizeSelect.value) || 10;
            const pageSize = all ? Math.max(total,1) : requested;
            const totalPages = total === 0 ? 1 : Math.max(1, Math.ceil(total / pageSize));

            page = Math.min(Math.max(page,1), totalPages);

            const startIndex = all ? 0 : (page - 1) * pageSize;
            const endIndex = all ? total : Math.min(startIndex + pageSize,total);

            rows.forEach(function(row,index){
                row.hidden = !(index >= startIndex && index < endIndex);
            });

            if(total === 0){
                info.textContent = '0 data';
            }else if(all){
                info.textContent = total + ' data';
            }else{
                info.textContent = (startIndex + 1) + '–' + endIndex + ' dari ' + total + ' data';
            }

            prev.disabled = all || total === 0 || page <= 1;
            next.disabled = all || total === 0 || page >= totalPages;
        }

        sizeSelect.addEventListener('change',function(){
            page = 1;
            applyPagination();
        });

        prev.addEventListener('click',function(){
            if(page > 1){
                page -= 1;
                applyPagination();
            }
        });

        next.addEventListener('click',function(){
            page += 1;
            applyPagination();
        });

        [
            root.querySelector('#spj26adm-residen-search'),
            root.querySelector('#spj26adm-residen-program-filter'),
            root.querySelector('#spj26adm-residen-status-filter')
        ].filter(Boolean).forEach(function(el){
            el.addEventListener(
                el.tagName === 'INPUT' ? 'input' : 'change',
                function(){ page = 1; },
                { capture:true }
            );
        });

        if(window.MutationObserver){
            new MutationObserver(function(){
                applyPagination();
            }).observe(body,{ childList:true });
        }

        applyPagination();

    })();


    function residentBoolean(value){
        if(value === undefined || value === null || value === ''){
            return true;
        }
        return value === true || value === 1 || value === '1' || value === 'true' || value === 'active' || value === 'yes';
    }

    function normalizeResident(item){
        item = item || {};
        return {
            ...item,
            id: item.id ?? item.user_id ?? item.ID ?? '',
            nim: item.nim ?? item.username ?? item.user_login ?? '',
            name: item.name ?? item.display_name ?? item.full_name ?? '',
            program: canonicalResidentProgram(
                item.program ?? item.study_program ?? item.prodi ?? ''
            ),
            active: residentBoolean(item.active ?? item.is_active ?? item.status),
            last_login: item.last_login ?? item.lastLogin ?? item.last_login_at ?? '',
            submissions: Number(item.submissions ?? item.submission_count ?? item.total_submissions ?? 0) || 0
        };
    }

    function extractResidents(data){
        const candidates = [
            data,
            data && data.residents,
            data && data.items,
            data && data.users,
            data && data.data,
            data && data.data && data.data.residents
        ];
        for(const candidate of candidates){
            if(Array.isArray(candidate)){
                return candidate.map(normalizeResident);
            }
        }
        return [];
    }

    function getAllowedPrograms(){

        const select =
            root.querySelector(
                '#spj26adm-residen-program'
            );

        if(!select){
            return [];
        }

        return Array.from(
            select.options
        )
        .map(
            option =>
                String(
                    option.value || ''
                ).trim()
        )
        .filter(Boolean);

    }


    function normalizeHeader(value){

        return String(
            value || ''
        )
        .trim()
        .toLowerCase()
        .replace(/[^a-z0-9]/g,'');

    }


    function findColumnIndex(
        headers,
        aliases
    ){

        const normalized =
            headers.map(
                normalizeHeader
            );

        for(
            let index = 0;
            index < normalized.length;
            index++
        ){

            if(
                aliases.includes(
                    normalized[index]
                )
            ){
                return index;
            }

        }

        return -1;

    }


    function maskPassword(password){

        const length =
            Math.max(
                8,
                Math.min(
                    String(
                        password || ''
                    ).length,
                    12
                )
            );

        return '•'.repeat(length);

    }


    function canonicalProgram(
        program,
        allowedPrograms
    ){

        const input =
            canonicalResidentProgram(
                program
            );

        const match =
            allowedPrograms.find(
                item =>
                    item.toLowerCase() ===
                    input.toLowerCase()
            );

        return match || input;

    }


    function validateImportedRows(rawRows){

        const allowedPrograms =
            getAllowedPrograms();

        const nimCounts = {};


        rawRows.forEach(
            function(row){

                const key =
                    row.nim.toLowerCase();

                if(key){

                    nimCounts[key] =
                        (nimCounts[key] || 0) + 1;

                }

            }
        );


        return rawRows.map(
            function(row){

                const errors = [];

                row.program =
                    canonicalProgram(
                        row.program,
                        allowedPrograms
                    );


                if(!row.nim){
                    errors.push(
                        'NIM wajib diisi'
                    );
                }

                if(!row.name){
                    errors.push(
                        'Nama Lengkap wajib diisi'
                    );
                }

                if(!row.program){
                    errors.push(
                        'Program PPDS wajib diisi'
                    );
                }else if(
                    allowedPrograms.length > 0 &&
                    !allowedPrograms.includes(
                        row.program
                    )
                ){
                    errors.push(
                        'Program PPDS tidak dikenali'
                    );
                }

                if(!row.password){
                    errors.push(
                        'Password Awal wajib diisi'
                    );
                }else if(
                    row.password.length < 8
                ){
                    errors.push(
                        'Password minimal 8 karakter'
                    );
                }

                if(
                    row.nim &&
                    nimCounts[
                        row.nim.toLowerCase()
                    ] > 1
                ){
                    errors.push(
                        'NIM duplikat di dalam file'
                    );
                }


                return {
                    rowNumber:
                        row.rowNumber,
                    nim:
                        row.nim,
                    name:
                        row.name,
                    program:
                        row.program,
                    password:
                        row.password,
                    validationErrors:
                        errors,
                    state:
                        errors.length
                        ? 'invalid'
                        : 'ready',
                    backendMessage:''
                };

            }
        );

    }


    function getImportCounts(){

        const counts = {
            total:importRows.length,
            ready:0,
            invalid:0,
            success:0,
            failed:0,
            processing:0
        };


        importRows.forEach(
            function(row){

                if(
                    Object.prototype.hasOwnProperty.call(
                        counts,
                        row.state
                    )
                ){
                    counts[row.state]++;
                }

            }
        );


        return counts;

    }

    async function processImportFile(file){

        resetBulkImport();


        if(!file){
            return;
        }


        if(
            !/\.(xlsx|xls)$/i.test(
                file.name || ''
            )
        ){

            showImportError(
                'File harus berformat .xlsx atau .xls.'
            );

            return;

        }


        if(
            file.size >
            5 * 1024 * 1024
        ){

            showImportError(
                'Ukuran file terlalu besar. Maksimal 5 MB.'
            );

            return;

        }


        if(
            typeof window.XLSX === 'undefined'
        ){

            showImportError(
                'Library Excel belum berhasil dimuat. Muat ulang halaman lalu coba kembali.'
            );

            return;

        }


        if(importFileName){
            importFileName.textContent =
                file.name;
        }


        try{

            const buffer =
                await file.arrayBuffer();


            const workbook =
                XLSX.read(
                    buffer,
                    {
                        type:'array'
                    }
                );


            if(
                !workbook.SheetNames ||
                workbook.SheetNames.length === 0
            ){

                throw new Error(
                    'Workbook tidak memiliki sheet.'
                );

            }


            const sheet =
                workbook.Sheets[
                    workbook.SheetNames[0]
                ];


            const rows =
                XLSX.utils.sheet_to_json(
                    sheet,
                    {
                        header:1,
                        defval:'',
                        raw:false
                    }
                );


            if(
                !Array.isArray(rows) ||
                rows.length < 2
            ){

                throw new Error(
                    'File belum berisi data mahasiswa.'
                );

            }


            const headers =
                rows[0].map(
                    value =>
                        String(
                            value || ''
                        ).trim()
                );


            const nimIndex =
                findColumnIndex(
                    headers,
                    [
                        'nim'
                    ]
                );

            const nameIndex =
                findColumnIndex(
                    headers,
                    [
                        'namalengkap',
                        'nama'
                    ]
                );

            const programIndex =
                findColumnIndex(
                    headers,
                    [
                        'programppds',
                        'program'
                    ]
                );

            const passwordIndex =
                findColumnIndex(
                    headers,
                    [
                        'passwordawal',
                        'password'
                    ]
                );


            const missingColumns = [];


            if(nimIndex < 0){
                missingColumns.push('NIM');
            }

            if(nameIndex < 0){
                missingColumns.push(
                    'Nama Lengkap'
                );
            }

            if(programIndex < 0){
                missingColumns.push(
                    'Program PPDS'
                );
            }

            if(passwordIndex < 0){
                missingColumns.push(
                    'Password Awal'
                );
            }


            if(missingColumns.length){

                throw new Error(
                    'Kolom wajib tidak ditemukan: ' +
                    missingColumns.join(', ') +
                    '. Gunakan template Excel SIPATUJU.'
                );

            }


            const rawRows = [];


            for(
                let i = 1;
                i < rows.length;
                i++
            ){

                const row =
                    rows[i] || [];


                const nim =
                    String(
                        row[nimIndex] || ''
                    ).trim();

                const name =
                    String(
                        row[nameIndex] || ''
                    ).trim();

                const program =
                    String(
                        row[programIndex] || ''
                    ).trim();

                const password =
                    String(
                        row[passwordIndex] || ''
                    ).trim();


                if(
                    !nim &&
                    !name &&
                    !program &&
                    !password
                ){
                    continue;
                }


                rawRows.push({
                    rowNumber:i + 1,
                    nim:nim,
                    name:name,
                    program:program,
                    password:password
                });

            }


            if(
                rawRows.length === 0
            ){

                throw new Error(
                    'Tidak ada data mahasiswa yang dapat diproses.'
                );

            }


            if(
                rawRows.length > 500
            ){

                throw new Error(
                    'Maksimal 500 baris mahasiswa per file.'
                );

            }


            importRows =
                validateImportedRows(
                    rawRows
                );


            clearImportError();
            renderImportState();


        }catch(error){

            importRows = [];

            showImportError(
                error.message ||
                'File Excel gagal dibaca.'
            );

            renderImportState();

        }

    }

            'spj26adm-detail-program',
            resident.program || '-'
        );

        setText(
            'spj26adm-detail-status',
            resident.active
            ? 'Aktif'
            : 'Nonaktif'
        );

        setText(
            'spj26adm-detail-login',
            resident.last_login ||
            'Belum pernah login'
        );

        setText(
            'spj26adm-detail-submissions',
            resident.submissions || 0
        );

        setText(
            'spj26adm-detail-login-state',
            resident.last_login
            ? 'Pernah login'
            : 'Belum pernah login'
        );


        if(detailModal){
            detailModal.hidden = false;
        }

    }


    function closeDetail(){

        if(detailModal){
            detailModal.hidden = true;
        }

    }


    if(detailClose){
        detailClose.addEventListener(
            'click',
            closeDetail
        );
    }

    if(detailCloseBottom){
        detailCloseBottom.addEventListener(
            'click',
            closeDetail
        );
    }

    if(detailModal){
        detailModal.addEventListener(
            'click',
            function(event){

                if(event.target === detailModal){
                    closeDetail();
                }

            }
        );
    }


    /* =====================================================
       ACTIVATE / DEACTIVATE
    ===================================================== */

    async function toggleResident(
        resident
    ){

        try{

            await backendRequest(
                'spj26_admin_toggle_resident',
                {
                    user_id:
                        resident.id,
                    active:
                        resident.active
                        ? '0'
                        : '1'
                }
            );


            showToast(
                resident.active
                ? 'Akun Mahasiswa berhasil dinonaktifkan.'
                : 'Akun Mahasiswa berhasil diaktifkan.'
            );


            await loadResidents();


        }catch(error){

            showToast(
                error.message ||
                'Perubahan status akun gagal.'
            );

        }

    }


    /* =====================================================
       LOGOUT
    ===================================================== */

    const logoutLink =
        root.querySelector(
            '#spj26adm-residen-logout'
        );


    function applyResidentConfig(){

        if(
            logoutLink &&
            window.spj26ResidentAdminConfig &&
            window.spj26ResidentAdminConfig.logoutUrl
        ){

            logoutLink.href =
                window.spj26ResidentAdminConfig.logoutUrl;

        }

    }


    document.addEventListener(
        'keydown',
        function(event){

            if(
                event.key === 'Escape' &&
                editModal &&
                !editModal.hidden
            ){
                closeEditResident();
            }
        }
    );


    /* =====================================================
       INITIAL
       Elementor HTML Widget dapat dieksekusi sebelum wp_footer
       mencetak window.spj26ResidentAdminConfig. Karena itu,
       tunggu sampai seluruh page selesai dimuat terlebih dahulu.
    ===================================================== */

    function initializeResidentPage(){

        applyResidentConfig();

        if(
            !window.spj26ResidentAdminConfig ||
            !window.spj26ResidentAdminConfig.ajaxUrl ||
            !window.spj26ResidentAdminConfig.nonce
        ){

            showToast(
                'Konfigurasi backend Manajemen Mahasiswa belum tersedia.'
            );

            console.error(
                'SIPATUJU Mahasiswa: spj26ResidentAdminConfig tidak ditemukan.'
            );

            return;

        }

        loadResidents();

    }


    if(
        document.readyState === 'complete'
    ){

        initializeResidentPage();

    }else{

        window.addEventListener(
            'load',
            initializeResidentPage,
            { once:true }
        );

    }

})();
</script>
