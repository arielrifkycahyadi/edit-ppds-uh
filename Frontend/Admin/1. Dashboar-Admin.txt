                count > 0

                : 'Hapus Terpilih';

        if(selectAllCheckbox){

            const visibleCheckboxes =
                getVisibleSubmissionCheckboxes();

            const selectedVisible =
                visibleCheckboxes.filter(function(checkbox){
                    return checkbox.checked;
                }).length;

            selectAllCheckbox.checked =
                visibleCheckboxes.length > 0 &&
                selectedVisible === visibleCheckboxes.length;

            selectAllCheckbox.indeterminate =
                selectedVisible > 0 &&
                selectedVisible < visibleCheckboxes.length;

            selectAllCheckbox.disabled =
                visibleCheckboxes.length === 0;
        }
    }

    /* =====================================================
       CLIENT-SIDE PAGINATION
       10 / 20 / 30 / Semua
    ===================================================== */
    (function setupPagination(){

        const body = root.querySelector('#spj26adm-submission-table');
        const sizeSelect = root.querySelector('#spj26adm-submission-page-size');
        const info = root.querySelector('#spj26adm-submission-page-info');
        const prev = root.querySelector('#spj26adm-submission-page-prev');
        const next = root.querySelector('#spj26adm-submission-page-next');

        if(!body || !sizeSelect || !info || !prev || !next){
            return;
        }

        let page = 1;

        function getDataRows(){
            return Array.from(body.children).filter(function(row){
                return !row.querySelector('.spj26adm-empty-state');
            });
        }

        function applyPagination(){

            const rows = getDataRows();
            const total = rows.length;
            const all = sizeSelect.value === 'all';
            const requested = Number(sizeSelect.value) || 10;
            const pageSize = all ? Math.max(total,1) : requested;
            const totalPages = total === 0 ? 1 : Math.max(1, Math.ceil(total / pageSize));

            page = Math.min(Math.max(page,1), totalPages);

            const startIndex = all ? 0 : (page - 1) * pageSize;
            const endIndex = all ? total : Math.min(startIndex + pageSize,total);

            rows.forEach(function(row,index){
                row.hidden = !(index >= startIndex && index < endIndex);
            });

            if(total === 0){
                info.textContent = '0 data';
            }else if(all){
                info.textContent = total + ' data';
            }else{
                info.textContent = (startIndex + 1) + '–' + endIndex + ' dari ' + total + ' data';
            }

            prev.disabled = all || total === 0 || page <= 1;
            next.disabled = all || total === 0 || page >= totalPages;

            updateBulkSelectionUI();
        }

        sizeSelect.addEventListener('change',function(){
            page = 1;
            applyPagination();
        });

        prev.addEventListener('click',function(){
            if(page > 1){
                page -= 1;
                applyPagination();
            }
        });

        next.addEventListener('click',function(){
            page += 1;
            applyPagination();
        });

        [
            root.querySelector('#spj26adm-search-input'),
            root.querySelector('#spj26adm-status-filter')
        ].filter(Boolean).forEach(function(el){
            el.addEventListener(
                el.tagName === 'INPUT' ? 'input' : 'change',
                function(){ page = 1; },
                { capture:true }
            );
        });

        if(window.MutationObserver){
            new MutationObserver(function(){
                applyPagination();
            }).observe(body,{ childList:true });
        }

        applyPagination();

    })();

    function renderDeleteSelectionList(){

        if(!deleteSelectionList){
            return;
        }

        deleteSelectionList.innerHTML = '';

            const row =
                createEl(

                )
            );

            row.appendChild(

                )
            );

            deleteSelectionList.appendChild(row);
        });
    }


    function openDeleteSubmissionModal(items){

        const normalized =
            Array.isArray(items)
            ? items
            : [items];

        deleteTargetSubmissions =
            normalized.filter(function(item){
                return (
                    item &&
                    item.id
                );
            });

        if(deleteTargetSubmissions.length === 0){
            showToast(
                'Tidak ada pengajuan yang dipilih.'
            );
            return;
        }

        const count =
            deleteTargetSubmissions.length;

        if(deleteQuestion){

            deleteQuestion.textContent =
                count === 1
                ? 'Apakah Anda yakin ingin menghapus data ini?'
                : 'Apakah Anda yakin ingin menghapus ' + count + ' data ini?';
        }

        const subtitle =
            root.querySelector(
                '#spj26adm-delete-subtitle'
            );

        if(subtitle){

            subtitle.textContent =
                count === 1
                ? (
                    (deleteTargetSubmissions[0].code || 'Pengajuan') +
                    ' • ' +
                    (deleteTargetSubmissions[0].resident_name || '-')
                )
                : (
                    count +
                    ' pengajuan dipilih untuk dihapus permanen'
                );
        }

        renderDeleteSelectionList();

        if(deleteConfirm){
            deleteConfirm.disabled = false;
            deleteConfirm.textContent = 'Ya Hapus';
        }

        if(deleteModal){
            deleteModal.hidden = false;
        }
    }


    function closeDeleteSubmissionModal(){

        if(deleteModal){
            deleteModal.hidden = true;
        }

        if(deleteConfirm){
            deleteConfirm.disabled = false;
            deleteConfirm.textContent = 'Ya Hapus';
        }

        deleteTargetSubmissions = [];

        if(deleteSelectionList){
            deleteSelectionList.innerHTML = '';
        }
    }


    [deleteClose,deleteCancel]
        .filter(Boolean)
        .forEach(function(button){

            button.addEventListener(
                'click',
                closeDeleteSubmissionModal
            );
        });


    if(deleteModal){

        deleteModal.addEventListener(
            'click',
            function(event){

                if(event.target === deleteModal){
                    closeDeleteSubmissionModal();
                }
            }
        );
    }

            }
        );
    }


    if(deleteConfirm){

        deleteConfirm.addEventListener(
            'click',
            async function(){

                if(
                    !deleteTargetSubmissions.length
                ){
                    showToast(
                        'Tidak ada pengajuan yang dipilih.'
                    );
                    return;
                }

                const ids =
                    deleteTargetSubmissions
                    .map(function(item){
                        return String(item.id);
                    })
                    .filter(Boolean);

                const deleteCount =
                    ids.length;

                deleteConfirm.disabled = true;
                deleteConfirm.textContent = 'Menghapus...';

                try{

                    await backendRequest(
                        'spj26_admin_delete_submission',
                        {
                            submission_ids:
                                ids.join(',')
                        }
                    );

                    closeDeleteSubmissionModal();

                    clearBulkSelection();

                    showToast(
                        deleteCount === 1
                        ? 'Pengajuan berhasil dihapus.'
                        : deleteCount + ' pengajuan berhasil dihapus.'
                    );

                        'Pengajuan gagal dihapus.'
                    );

                    deleteConfirm.textContent =
                        'Ya Hapus';

                    deleteConfirm.disabled =
                        false;
                }
            }
        );
    }

    /* =====================================================
       BACKEND REQUEST
    ===================================================== */

    async function backendRequest(
        action,
        payload
    ){

        if(
            !window.spj26AdminDashboardConfig ||
            !window.spj26AdminDashboardConfig.ajaxUrl ||
            !window.spj26AdminDashboardConfig.nonce
        ){

            throw new Error(
                'Konfigurasi backend dashboard belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            window.spj26AdminDashboardConfig.nonce
        );


        Object.keys(
            payload || {}
        ).forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                window.spj26AdminDashboardConfig.ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result ||
            result.success !== true
        ){

            const message =
                result &&
                result.data &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

            throw new Error(message);

        }


        return result.data;

    }


    async function loadDashboard(){

        /*
         * Default selalu kosong.
         * Tidak ada data contoh/fallback.
         */

        renderCounts();
        renderSubmissions();
        renderActivities();
        renderReviewers();


        if(
            !window.spj26AdminDashboardConfig
        ){
            return;
        }


        try{

            const data =
                await backendRequest(
                    'spj26_admin_dashboard_data',
                    {}
                );


            if(dashboardAlert){
                dashboardAlert.hidden = true;
                dashboardAlert.textContent = '';
            }


            dashboardData = {
                counts:
                    data.counts || {
                        waiting:0,
                        progress:0,
                        done:0,
                        reviewers:0
                    },
                submissions:
                    Array.isArray(data.submissions)
                    ? data.submissions
                    : [],
                activities:
                    Array.isArray(data.activities)
                    ? data.activities
                    : [],
                reviewers:
                    Array.isArray(data.reviewers)
                    ? data.reviewers
                    : []
            };


            const validSubmissionIds =
                new Set(
                    dashboardData.submissions.map(function(item){
                        return String(item.id);
                    })
                );

            Array.from(
                selectedSubmissionIds
            ).forEach(function(id){

                if(!validSubmissionIds.has(id)){
                    selectedSubmissionIds.delete(id);
                }
            });

            renderCounts();
            renderSubmissions();
            renderActivities();
            renderReviewers();


        }catch(error){

            /*
             * Jika backend gagal, data palsu tidak ditampilkan.
             * Error ditampilkan agar dashboard tidak terlihat seperti
             * benar-benar memiliki nilai 0.
             */
            console.error(
                'SIPATUJU Dashboard:',
                error
            );

            if(dashboardAlert){

                dashboardAlert.textContent =
                    error.message ||
                    'Data dashboard gagal dimuat dari backend SIPATUJU.';

                dashboardAlert.hidden =
                    false;
            }

        }

    }


    if(confirmAssignment){

        confirmAssignment.addEventListener(
            'click',
            async function(){

                const reviewerId =
                    reviewerSelect
                    ? reviewerSelect.value
                    : '';


                if(
                    !activeSubmissionId ||
                    !reviewerId
                ){

                    showToast(
                        'Pilih reviewer terlebih dahulu.'
                    );

                    return;

                }


                confirmAssignment.disabled =
                    true;


                try{

                    await backendRequest(
                        'spj26_admin_assign_reviewer',
                        {
                            submission_id:
                                activeSubmissionId,
                            reviewer_id:
                                reviewerId
                        }
                    );


                    closeAssignmentModal();

                    showToast(
                        'Reviewer berhasil ditugaskan.'
                    );


                    await loadDashboard();


                }catch(error){

                    showToast(
                        error.message ||
                        'Penugasan reviewer gagal.'
                    );

                }finally{

                    confirmAssignment.disabled =
                        false;

                }

            }
        );

    }


    /* =====================================================
       CONFIG / LOGOUT
    ===================================================== */

    function applyDashboardConfig(){

        const config =
            window.spj26AdminDashboardConfig;


        if(!config){
            return false;
        }


        const logoutLink =
            root.querySelector(
                '#spj26adm-logout'
            );


        if(
            logoutLink &&
            config.logoutUrl
        ){

            logoutLink.href =
                config.logoutUrl;

        }


        return (
            !!config.ajaxUrl &&
            !!config.nonce
        );

    }


    document.addEventListener(
        'keydown',
        function(event){

            if(event.key !== 'Escape'){
                return;
            }

            if(
                deleteModal &&
                !deleteModal.hidden
            ){
                closeDeleteSubmissionModal();
                return;
            }

            if(
                detailModal &&
                !detailModal.hidden
            ){
                closeSubmissionDetail();
                return;
            }

            if(
                modal &&
                !modal.hidden
            ){
                closeAssignmentModal();
            }
        }
    );


    /* =====================================================
       INITIAL
    ===================================================== */

    function initializeAdminDashboard(){

        /*
         * Config backend dicetak melalui wp_footer.
         * Elementor HTML Widget dapat dieksekusi sebelum config tersebut ada,
         * jadi dashboard harus menunggu window.load.
         */
        if(
            !applyDashboardConfig()
        ){

            console.error(
                'SIPATUJU Admin: spj26AdminDashboardConfig tidak ditemukan.'
            );

            if(dashboardAlert){

                dashboardAlert.textContent =
                    'Konfigurasi backend Dashboard Admin belum tersedia. Pastikan snippet 000 — 02 dan 000 — 03 aktif.';

                dashboardAlert.hidden =
                    false;
            }

            renderCounts();
            renderSubmissions();
            renderActivities();
            renderReviewers();

            return;
        }


        loadDashboard();

    }


    if(
        document.readyState === 'complete'
    ){

        initializeAdminDashboard();

    }else{

        window.addEventListener(
            'load',
            initializeAdminDashboard,
            { once:true }
        );

    }

})();
