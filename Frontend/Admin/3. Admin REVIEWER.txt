    function getReviewerConfig(){

    async function backendRequest(
        action,
        payload
    ){

        if(
            !getReviewerConfig() ||
            !getReviewerConfig().ajaxUrl ||
            !getReviewerConfig().nonce
        ){

            throw new Error(
                'Konfigurasi backend Manajemen Reviewer belum tersedia.'
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            'action',
            action
        );

        body.append(
            'nonce',
            getReviewerConfig().nonce
        );


        Object.keys(
            payload || {}
        ).forEach(function(key){

            body.append(
                key,
                payload[key]
            );

        });


        const response =
            await fetch(
                getReviewerConfig().ajaxUrl,
                {
                    method:'POST',
                    credentials:'same-origin',
                    headers:{
                        'Content-Type':
                            'application/x-www-form-urlencoded; charset=UTF-8'
                    },
                    body:
                        body.toString()
                }
            );


        const result =
            await response.json();


        if(
            !result ||
            result.success !== true
        ){

            const message =
                result &&
                result.data &&
                result.data.message
                ? result.data.message
                : 'Permintaan ke server gagal.';

            throw new Error(message);

        }


        return result.data;

    }

    function renderUnitFilter(){

            .sort(

            );


        unitFilter.innerHTML = '';


        const all =
            document.createElement('option');

        all.value = 'all';
        all.textContent =
            'Semua Unit / Bagian';

        unitFilter.appendChild(all);


        units.forEach(function(unit){

            const option =
                document.createElement('option');

            option.value = unit;
            option.textContent = unit;

            unitFilter.appendChild(option);

        });


        if(
            selected &&
            (
                selected === 'all' ||
                units.includes(selected)
            )
        ){
            unitFilter.value = selected;
        }

    }


    /* =====================================================

    function getFilteredReviewers(){

        const query =
            searchInput
            ? searchInput.value.trim().toLowerCase()
            : '';

        const unit =
            unitFilter
            ? unitFilter.value
            : 'all';

        const status =
            statusFilter
            ? statusFilter.value
            : 'all';


        return reviewers.filter(function(item){

            const text =
                [
                    item.username,
                    item.name,
                    item.unit
                ]
                .filter(Boolean)
                .join(' ')
                .toLowerCase();


            const queryMatch =
                !query ||
                text.includes(query);


            const unitMatch =
                unit === 'all' ||
                item.unit === unit;


            let statusMatch = true;


            if(status === 'active'){

                statusMatch =
                    item.active === true;

            }else if(status === 'inactive'){

                statusMatch =
                    item.active === false;

            }else if(status === 'never'){

                statusMatch =
                    item.active === true &&
                    !item.last_login;

            }


            return (
                queryMatch &&
                unitMatch &&
                statusMatch
            );

        });

    }

    /* =====================================================
       CLIENT-SIDE PAGINATION
       10 / 20 / 30 / Semua
    ===================================================== */
    (function setupPagination(){

        const body = root.querySelector('#spj26adm-reviewer-table-body');
        const sizeSelect = root.querySelector('#spj26adm-reviewer-page-size');
        const info = root.querySelector('#spj26adm-reviewer-page-info');
        const prev = root.querySelector('#spj26adm-reviewer-page-prev');
        const next = root.querySelector('#spj26adm-reviewer-page-next');

        if(!body || !sizeSelect || !info || !prev || !next){
            return;
        }

        let page = 1;

        function getDataRows(){
            return Array.from(body.children).filter(function(row){
                return !row.querySelector('.spj26adm-empty-state');
            });
        }

        function applyPagination(){

            const rows = getDataRows();
            const total = rows.length;
            const all = sizeSelect.value === 'all';
            const requested = Number(sizeSelect.value) || 10;
            const pageSize = all ? Math.max(total,1) : requested;
            const totalPages = total === 0 ? 1 : Math.max(1, Math.ceil(total / pageSize));

            page = Math.min(Math.max(page,1), totalPages);

            const startIndex = all ? 0 : (page - 1) * pageSize;
            const endIndex = all ? total : Math.min(startIndex + pageSize,total);

            rows.forEach(function(row,index){
                row.hidden = !(index >= startIndex && index < endIndex);
            });

            if(total === 0){
                info.textContent = '0 data';
            }else if(all){
                info.textContent = total + ' data';
            }else{
                info.textContent = (startIndex + 1) + '–' + endIndex + ' dari ' + total + ' data';
            }

            prev.disabled = all || total === 0 || page <= 1;
            next.disabled = all || total === 0 || page >= totalPages;
        }

        sizeSelect.addEventListener('change',function(){
            page = 1;
            applyPagination();
        });

        prev.addEventListener('click',function(){
            if(page > 1){
                page -= 1;
                applyPagination();
            }
        });

        next.addEventListener('click',function(){
            page += 1;
            applyPagination();
        });

        [
            root.querySelector('#spj26adm-reviewer-search'),
            root.querySelector('#spj26adm-reviewer-unit-filter'),
            root.querySelector('#spj26adm-reviewer-status-filter')
        ].filter(Boolean).forEach(function(el){
            el.addEventListener(
                el.tagName === 'INPUT' ? 'input' : 'change',
                function(){ page = 1; },
                { capture:true }
            );
        });

        if(window.MutationObserver){
            new MutationObserver(function(){
                applyPagination();
            }).observe(body,{ childList:true });
        }

        applyPagination();

    })();


    function reviewerBoolean(value){
        if(value === undefined || value === null || value === ''){
            return true;
        }
        return value === true || value === 1 || value === '1' || value === 'true' || value === 'active' || value === 'yes';
    }

    function normalizeReviewer(item){
        item = item || {};
        return {
            ...item,
            id: item.id ?? item.user_id ?? item.ID ?? '',
            username: item.username ?? item.user_login ?? item.login ?? '',
            name: item.name ?? item.display_name ?? item.full_name ?? '',
            unit: item.unit ?? item.department ?? item.section ?? item.bagian ?? '',
            active: reviewerBoolean(item.active ?? item.is_active ?? item.status),
            last_login: item.last_login ?? item.lastLogin ?? item.last_login_at ?? '',
            active_submissions: Number(item.active_submissions ?? item.active_reviews ?? item.active_work ?? 0) || 0,
            completed_reviews: Number(item.completed_reviews ?? item.done_reviews ?? item.completed ?? 0) || 0
        };
    }

    function extractReviewers(data){
        const candidates = [
            data,
            data && data.reviewers,
            data && data.items,
            data && data.users,
            data && data.data,
            data && data.data && data.data.reviewers
        ];

        for(const candidate of candidates){
            if(Array.isArray(candidate)){
                return candidate.map(normalizeReviewer);
            }
        }

        return [];
    }

    async function loadReviewers(){

        reviewers = [];

        renderMetrics();
        renderUnitFilter();
        renderTable();

            renderUnitFilter();

                'SIPATUJU Reviewer:',
                error
            );

            showToast(
                error.message ||
                'Data Reviewer gagal dimuat.'
            );

        }

    }

    function clearForm(){

        [
            'spj26adm-reviewer-username',
            'spj26adm-reviewer-name',
            'spj26adm-reviewer-unit',
            'spj26adm-reviewer-password',
            'spj26adm-reviewer-password-confirm'
        ].forEach(function(id){

            const el =
                root.querySelector(
                    '#' + id
                );

            if(el){
                el.value = '';
            }

        });


        root
            .querySelectorAll(
                '[data-password-toggle]'
            )
            .forEach(function(button){

                const targetId =
                    button.getAttribute(
                        'data-password-toggle'
                    );

                const input =
                    root.querySelector(
                        '#' + targetId
                    );

                if(input){
                    input.type = 'password';
                }

                const icon =
                    button.querySelector(
                        '.material-symbols-outlined'
                    );

                if(icon){
                    icon.textContent =
                        'visibility';
                }

                button.setAttribute(
                    'aria-pressed',
                    'false'
                );

            });


        if(formError){
            formError.classList.remove('show');
            formError.textContent = '';
        }

    }


    function openAddModal(){

        clearForm();

        if(addModal){
            addModal.hidden = false;
        }

    }

        if(addModal){
            addModal.hidden = true;
        }

    }

        if(!formError) return;

        formError.classList.add('show');

            'click',
            async function(){

                const username =
                    getInput(
                        'spj26adm-reviewer-username'
                    );

                const name =
                    getInput(
                        'spj26adm-reviewer-name'
                    );

                const unit =
                    getInput(
                        'spj26adm-reviewer-unit'
                    );

                const password =
                    getInput(
                        'spj26adm-reviewer-password'
                    );

                const confirmPassword =
                    getInput(
                        'spj26adm-reviewer-password-confirm'
                    );


                if(
                    !username ||
                    !name ||
                    !password ||
                    !confirmPassword
                ){

                    showFormError(
                        'Username, nama lengkap, password, dan konfirmasi password wajib diisi.'
                    );

                    return;

                }


                if(
                    password.length < 8
                ){

                    showFormError(
                        'Password minimal 8 karakter.'
                    );

                    return;

                }


                if(
                    password !==
                    confirmPassword
                ){

                    showFormError(
                        'Konfirmasi password tidak sama.'
                    );

                    return;

                }


                saveButton.disabled = true;


                try{

                    await backendRequest(
                        'spj26_admin_create_reviewer',
                        {
                            username:username,
                            name:name,
                            unit:unit,
                            password:password
                        }
                    );

                    showToast(

                    );


                    await loadReviewers();


                }catch(error){

                    showFormError(
                        error.message ||
                        'Gagal membuat akun Reviewer.'
                    );

                }finally{

                    saveButton.disabled = false;

                }

            }
        );

    }

    async function toggleReviewer(
        reviewer
    ){

        try{

            await backendRequest(
                'spj26_admin_toggle_reviewer',
                {
                    user_id:
                        reviewer.id,
                    active:
                        reviewer.active
                        ? '0'
                        : '1'
                }
            );


            showToast(
                reviewer.active
                ? 'Akun Reviewer berhasil dinonaktifkan.'
                : 'Akun Reviewer berhasil diaktifkan.'
            );


            await loadReviewers();


        }catch(error){

            showToast(
                error.message ||
                'Perubahan status akun gagal.'
            );

        }

    }

    function initializeReviewerPage(){

        applyReviewerConfig();


        const config = getReviewerConfig();

            !config ||
            !config.ajaxUrl ||
            !config.nonce
        ){

                'SIPATUJU Reviewer: spj26ReviewerAdminConfig tidak ditemukan.'
            );

            showToast(
                'Konfigurasi backend Manajemen Reviewer belum tersedia.'
            );

        }


        loadReviewers();

    }

        document.readyState === 'complete'
