    function buildResultsNotificationSnapshot(items) {
        const snapshot = {};
        const list = Array.isArray(items) ? items : [];

        list.forEach(function (item) {
            if (!item) return;

            const key = String(item.id || item.code || '').trim();
            if (!key) return;

            const journals = Array.isArray(item.journals)
                ? item.journals.map(function (journal) {
                    return {
                        order: journal && journal.order ? String(journal.order) : '',
                        decision: journal && journal.decision ? String(journal.decision) : '',
                        comment: journal && journal.comment ? String(journal.comment) : '',
                        updated_at: journal && journal.updated_at ? String(journal.updated_at) : ''
                    };
                })
                : [];

            snapshot[key] = JSON.stringify({
                status: item.status ? String(item.status) : '',
                status_label: item.status_label ? String(item.status_label) : '',
                updated_at: item.updated_at ? String(item.updated_at) : '',
                completed_at: item.completed_at ? String(item.completed_at) : '',
                journals: journals
            });
        });

        return snapshot;
    }

        let unreadCount = 0;

        Object.keys(currentSnapshot).forEach(function (key) {
            /*
             * Record baru tidak dihitung sebagai perubahan status.
             * Notifikasi hanya untuk pengajuan yang sebelumnya sudah dikenal,
             * lalu mengalami perubahan status/detail pemeriksaan.
             */
            if (!Object.prototype.hasOwnProperty.call(seenSnapshot, key)) {
                return;
            }

            if (seenSnapshot[key] !== currentSnapshot[key]) {
                unreadCount += 1;
            }
        });

        return unreadCount;
    }

    function renderResultsUpdateNotification(count) {
        const unreadCount = Math.max(0, Number(count) || 0);
        const hasUnread = unreadCount > 0;

        if (resultsNavAlert) {
            resultsNavAlert.hidden = !hasUnread;
        }

        if (resultsNavCount) {
            resultsNavCount.hidden = !hasUnread;
            resultsNavCount.textContent = hasUnread ? String(unreadCount) : '0';
        }
    }

    function markResultsAsSeen(items) {
        writeSeenResultsNotificationSnapshot(items);
        renderResultsUpdateNotification(0);
    }

    function renderFilters() {
        if (!filterBar) return;

        const counts = getCounts();
        filterBar.innerHTML = '';

        FILTER_CONFIG.forEach(function (filter) {
            const button = document.createElement('button');
            button.type = 'button';
            button.className = 'spj26res-filter-chip' + (activeFilter === filter.key ? ' is-active' : '');
            button.dataset.filter = filter.key;

            const dot = createEl('span', 'spj26res-filter-chip-dot');
            dot.textContent = filter.label;

            const count = createEl('span', 'spj26res-filter-chip-count', '(' + String(counts[filter.key] || 0) + ')');

            button.appendChild(dot);
            button.appendChild(count);

            button.addEventListener('click', function () {
                activeFilter = filter.key;
                currentPage = 1;
                renderFilters();
                applySearchAndRender();
            });

            filterBar.appendChild(button);
        });
    }

    function matchesSearch(item, query) {
        if (!query) return true;

        const haystack = [
            item.code,
            item.article_title,
            item.article_type,
            item.supervisor_name,
            item.submitted_at,
            item.updated_at,
            item.completed_at,
            item.status_label,
            (item.journals || []).map(function (journal) {
                return [journal.name, journal.issn, journal.quartile, journal.comment].join(' ');
            }).join(' ')
        ].join(' ').toLowerCase();

        return haystack.indexOf(query.toLowerCase()) !== -1;
    }

    function getEffectivePageSize() {
        if (itemsPerPage === 'all') {
            return Math.max(filteredSubmissions.length, 1);
        }

        return Math.max(1, Number(itemsPerPage) || 10);
    }

    function applySearchAndRender() {
        const query = searchInput ? String(searchInput.value || '').trim() : '';

        filteredSubmissions = submissions.filter(function (item) {
            const status = getDisplayStatus(item);
            const passFilter = activeFilter === 'all' || status.key === activeFilter;
            const passSearch = matchesSearch(item, query);
            return passFilter && passSearch;
        });

        const pageSize = getEffectivePageSize();
        const totalPages = Math.max(1, Math.ceil(filteredSubmissions.length / pageSize));
        if (currentPage > totalPages) {
            currentPage = totalPages;
        }
        if (currentPage < 1) {
            currentPage = 1;
        }

        renderTable();
        renderPagination();
    }

    function renderEmpty() {
        if (!tableBody) return;
        tableBody.innerHTML = '';

        const empty = createEl('div', 'spj26res-empty');
        empty.appendChild(createEl('span', 'material-symbols-outlined', 'description'));
        empty.appendChild(createEl('div', 'spj26res-empty-title', 'Belum ada data yang sesuai'));
        empty.appendChild(createEl('div', 'spj26res-empty-text', 'Pengajuan yang Anda kirim akan tampil di sini. Gunakan filter lain atau ubah kata kunci pencarian jika diperlukan.'));

        tableBody.appendChild(empty);
    }

    function buildCodeCell(item) {
        const wrapper = createEl('div', 'spj26res-table-cell');
        wrapper.appendChild(createEl('div', 'spj26res-code', item.code || '-'));

        const chips = createEl('div', 'spj26res-code-chips');

        if (item.article_type) {
            chips.appendChild(createEl('span', 'spj26res-mini-chip', item.article_type));
        }

        const quartiles = getQuartileSummary(item);
        if (quartiles) {
            chips.appendChild(createEl('span', 'spj26res-mini-chip', quartiles));
        }

        if (chips.childNodes.length > 0) {
            wrapper.appendChild(chips);
        }

        const dates = createEl('div', 'spj26res-code-dates');

        if (item.submitted_at) {
            const submit = createEl('div', 'spj26res-table-cell-note');
            submit.innerHTML = '<strong>Submit:</strong> ' + escapeHtml(item.submitted_at);
            dates.appendChild(submit);
        }

        if (item.completed_at) {
            const completed = createEl('div', 'spj26res-table-cell-note');
            completed.innerHTML = '<strong>Selesai:</strong> ' + escapeHtml(item.completed_at);
            dates.appendChild(completed);
        } else if (item.updated_at) {
            const updated = createEl('div', 'spj26res-table-cell-note');
            updated.innerHTML = '<strong>Update:</strong> ' + escapeHtml(item.updated_at);
            dates.appendChild(updated);
        }

        if (dates.childNodes.length > 0) {
            wrapper.appendChild(dates);
        }

        return wrapper;
    }

    function buildTitleCell(item) {
        const wrapper = createEl('div', 'spj26res-table-cell');
        wrapper.appendChild(createEl('div', 'spj26res-title', item.article_title || '-'));


        if (item.article_abstract) {
            const abstractSnippet = item.article_abstract.length > 140
                ? item.article_abstract.slice(0, 140) + '...'
                : item.article_abstract;
            wrapper.appendChild(createEl('div', 'spj26res-table-cell-note', abstractSnippet));
        }

        return wrapper;
    }

    function buildPeopleCell(item) {
        const wrapper = createEl('div', 'spj26res-table-cell');

        wrapper.appendChild(createEl('div', 'spj26res-person-name', item.supervisor_name || '-'));

        return wrapper;
    }

    function buildStatusCell(item) {
        const status = getDisplayStatus(item);
        const wrapper = createEl('div', 'spj26res-table-cell');
        wrapper.appendChild(createEl('span', 'spj26res-status-pill ' + status.className, status.label));
        return wrapper;
    }

    function buildActionButton(text, className, onClick, iconName) {
        const button = createEl('button', 'spj26res-btn ' + (className || ''));
        button.type = 'button';

        if (iconName) {
            button.appendChild(createEl('span', 'material-symbols-outlined', iconName));
        }
        button.appendChild(createEl('span', '', text));
        button.addEventListener('click', onClick);
        return button;
    }

    function resolveLoaUrl(item) {
        const config = getPortalConfig();

        const directUrl =
            (item && (item.loa_url || item.loaUrl || item.loa_download_url || item.loaDownloadUrl)) ||
            '';

        if (directUrl) {
            return String(directUrl);
        }

        if (config.loaUrlTemplate) {
            return String(config.loaUrlTemplate)
                .replaceAll('{id}', encodeURIComponent(String(item && item.id ? item.id : '')))
                .replaceAll('{code}', encodeURIComponent(String(item && item.code ? item.code : '')));
        }

        if (config.loaDownloadUrl) {
            try {
                const url = new URL(String(config.loaDownloadUrl), window.location.origin);
                if (item && item.id) {
                    url.searchParams.set('submission_id', String(item.id));
                }
                if (item && item.code) {
                    url.searchParams.set('code', String(item.code));
                }
                return url.toString();
            } catch (error) {
                console.error('SIPATUJU Residen - URL LoA tidak valid:', error);
            }
        }

        return '';
    }

    function goToLoaDownload(item) {
        const loaUrl = resolveLoaUrl(item);

        if (!loaUrl) {
            showToast('URL halaman Unduh Surat belum dikonfigurasi pada backend.');
            return;
        }

        window.location.assign(loaUrl);
    }

    function buildActionsCell(item) {
        const wrapper = createEl('div', 'spj26res-table-cell');
        const actions = createEl('div', 'spj26res-row-actions');
        const status = getDisplayStatus(item);

        if (status.key === 'completed') {
            const loaUrl = resolveLoaUrl(item);
            const loaAvailable =
                item &&
                (item.loa_available === true || item.loa_available === 1 || item.loa_available === '1') &&
                !!loaUrl;

            if (loaAvailable) {
                actions.appendChild(
                    buildActionButton('Unduh Surat', 'spj26res-btn-primary', function () {
                        goToLoaDownload(item);
                    }, 'download')
                );
            } else {
                const unavailableButton =
                    buildActionButton('LoA Belum Tersedia', '', function () {
                        showToast('LoA belum tersedia untuk artikel ini.');
                    }, 'hourglass_empty');

                unavailableButton.disabled = true;
                unavailableButton.setAttribute('aria-disabled', 'true');
                unavailableButton.style.cursor = 'not-allowed';
                unavailableButton.style.opacity = '0.58';

                actions.appendChild(unavailableButton);
            }
        } else if (status.key === 'not_eligible') {
            actions.appendChild(
                buildActionButton('Lihat Detail', '', function () {
                    openDetailModal(item, true);
                })
            );
        } else {
            actions.appendChild(
                buildActionButton('Lihat Detail', '', function () {
                    openDetailModal(item, false);
                })
            );
        }

        wrapper.appendChild(actions);
        return wrapper;
    }

    function renderTable() {
        if (!tableBody) return;

        if (filteredSubmissions.length === 0) {
            renderEmpty();
            return;
        }

        const pageSize = getEffectivePageSize();
        const startIndex = (currentPage - 1) * pageSize;
        const endIndex = startIndex + pageSize;
        const pageItems = filteredSubmissions.slice(startIndex, endIndex);

        tableBody.innerHTML = '';

        pageItems.forEach(function (item) {
            const row = createEl('div', 'spj26res-table-row');
            row.appendChild(buildCodeCell(item));
            row.appendChild(buildTitleCell(item));
            row.appendChild(buildPeopleCell(item));
            row.appendChild(buildStatusCell(item));
            row.appendChild(buildActionsCell(item));
            tableBody.appendChild(row);
        });

    }

    function renderPagination() {
        const pageSize = getEffectivePageSize();
        const totalPages = Math.max(1, Math.ceil(filteredSubmissions.length / pageSize));

        if (pageIndicator) {
            pageIndicator.textContent = String(currentPage);
        }

        if (prevButton) {
            prevButton.disabled = currentPage <= 1;
        }

        if (nextButton) {
            nextButton.disabled = currentPage >= totalPages;
        }
    }

        if (!detailModal || !detailContent) return;

        const status = getDisplayStatus(item);
        detailTitle.textContent = item.code || 'Detail Artikel';

        const journals = Array.isArray(item.journals) ? item.journals : [];

                '<div class="spj26res-detail-box">',
                    '<div class="spj26res-detail-label">Judul Artikel</div>',
                    '<div class="spj26res-detail-value">' + escapeHtml(item.article_title || '-') + '</div>',
                '</div>',
                '<div class="spj26res-detail-box">',
                    '<div class="spj26res-detail-label">Jenis Artikel</div>',
                    '<div class="spj26res-detail-value">' + escapeHtml(item.article_type || '-') + '</div>',
                '</div>',
                '<div class="spj26res-detail-box spj26res-detail-box-full">',
                    '<div class="spj26res-detail-label">Abstrak</div>',
                    '<div class="spj26res-detail-value">' + escapeHtml(item.article_abstract || '-') + '</div>',

                '<div class="spj26res-detail-box">',
                    '<div class="spj26res-detail-label">Tanggal Submit</div>',

                : journal.decision === 'tidak_layak'

    function closeDetailModal() {
        if (!detailModal) return;
        detailModal.classList.remove('open');
        detailModal.setAttribute('aria-hidden', 'true');
        document.body.style.overflow = '';
    }

    if (detailBackdrop) {
        detailBackdrop.addEventListener('click', closeDetailModal);
    }

    if (detailClose) {
        detailClose.addEventListener('click', closeDetailModal);
    }

    document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape') {
            closeDetailModal();
        }
    });

    let resultsRequestInFlight = false;
    let resultsRefreshTimer = null;


    function showSubmittedConfirmationFromQuery() {

        const params =
            new URLSearchParams(
                window.location.search
            );


        const submittedCode =
            String(
                params.get('submitted')
                ||
                ''
            ).trim();


        if (!submittedCode) {
            return;
        }


        showToast(
            'Pengajuan '
            +
            submittedCode
            +
            ' berhasil dikirim dan sudah masuk ke Hasil Pemeriksaan.'
        );


        params.delete(
            'submitted'
        );


        const cleanQuery =
            params.toString();


        const cleanUrl =
            window.location.pathname
            +
            (
                cleanQuery
                ?
                '?'
                +
                cleanQuery
                :
                ''
            )
            +
            window.location.hash;


        window.history.replaceState(
            {},
            document.title,
            cleanUrl
        );

    }


    async function loadResults(options) {
        const settings = options || {};

        if (resultsRequestInFlight) {
            return;
        }

        resultsRequestInFlight = true;

        try {
            const data = await backendRequest('spj26_resident_results_data', {});

            submissions = Array.isArray(data.submissions) ? data.submissions : [];
            updatedText.textContent = formatUpdatedText();
            renderFilters();
            applySearchAndRender();

            /*
             * Halaman Hasil Pemeriksaan adalah titik "dibaca".
             * Setelah data berhasil ditampilkan, kondisi terbaru disimpan
             * sebagai baseline dan badge notifikasi langsung dibersihkan.
             */
            markResultsAsSeen(submissions);
        } catch (error) {
            console.error('SIPATUJU Residen - Hasil Pemeriksaan:', error);

            if (!settings.silent) {
                showToast(error.message || 'Data Hasil Pemeriksaan gagal dimuat.');
            }
        } finally {
            resultsRequestInFlight = false;
        }
    }

    function startAutoRefresh() {
        if (resultsRefreshTimer) {
            window.clearInterval(resultsRefreshTimer);
        }

        resultsRefreshTimer = window.setInterval(function () {
            loadResults({ silent: true });
        }, 30000);
    }

    if (searchInput) {
        searchInput.addEventListener('input', function () {
            currentPage = 1;
            applySearchAndRender();
        });
    }

    if (pageSizeSelect) {
        pageSizeSelect.addEventListener('change', function () {
            const selectedValue = String(this.value || '10');

            itemsPerPage = selectedValue === 'all'
                ? 'all'
                : Number(selectedValue);

            currentPage = 1;
            applySearchAndRender();
        });
    }

    if (prevButton) {
        prevButton.addEventListener('click', function () {
            if (currentPage > 1) {
                currentPage -= 1;
                renderTable();
                renderPagination();
            }
        });
    }

    if (nextButton) {
        nextButton.addEventListener('click', function () {
            const pageSize = getEffectivePageSize();
            const totalPages = Math.max(1, Math.ceil(filteredSubmissions.length / pageSize));
            if (currentPage < totalPages) {
                currentPage += 1;
                renderTable();
                renderPagination();
            }
        });
    }

    function initializeResultsPage() {
        if (!window.spj26ResidentResultsConfig || !window.spj26ResidentResultsConfig.ajaxUrl || !window.spj26ResidentResultsConfig.nonce) {
            console.error('SIPATUJU Residen: spj26ResidentResultsConfig tidak ditemukan.');
            showToast('Konfigurasi backend Hasil Pemeriksaan belum tersedia.');
            return;
        }

        applyPortalConfig();
        loadResults();
        startAutoRefresh();
    }

    window.addEventListener('pagehide', function () {
        if (resultsRefreshTimer) {
            window.clearInterval(resultsRefreshTimer);
            resultsRefreshTimer = null;
        }
    });

    if (document.readyState === 'complete') {
        initializeResultsPage();
    } else {
        window.addEventListener('load', initializeResultsPage, { once: true });
    }
})();
