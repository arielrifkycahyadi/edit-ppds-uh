    function getResultsNotificationStorageKey() {
        const config = window.spj26ResidentResultsConfig || {};
        const identity = String(config.nim || config.name || 'resident')
            .trim()
            .replace(/[^a-zA-Z0-9_-]/g, '_');

        return 'spj26res_results_seen_v2_' + identity;
    }

    function buildResultsNotificationSnapshot(items) {
        const snapshot = {};
        const list = Array.isArray(items) ? items : [];

        list.forEach(function (item) {
            if (!item) return;

            const key = String(item.id || item.code || '').trim();
            if (!key) return;

            const journals = Array.isArray(item.journals)
                ? item.journals.map(function (journal) {
                    return {
                        order: journal && journal.order ? String(journal.order) : '',
                        decision: journal && journal.decision ? String(journal.decision) : '',
                        comment: journal && journal.comment ? String(journal.comment) : '',
                        updated_at: journal && journal.updated_at ? String(journal.updated_at) : ''
                    };
                })
                : [];

            snapshot[key] = JSON.stringify({
                status: item.status ? String(item.status) : '',
                status_label: item.status_label ? String(item.status_label) : '',
                updated_at: item.updated_at ? String(item.updated_at) : '',
                completed_at: item.completed_at ? String(item.completed_at) : '',
                journals: journals
            });
        });

        return snapshot;
    }

    function readSeenResultsNotificationSnapshot() {
        try {
            const raw = window.localStorage.getItem(
                getResultsNotificationStorageKey()
            );

            if (!raw) return null;

            const parsed = JSON.parse(raw);
            return parsed && typeof parsed === 'object' ? parsed : null;
        } catch (error) {
            console.warn('SIPATUJU Residen - snapshot notifikasi tidak dapat dibaca:', error);
            return null;
        }
    }

    function writeSeenResultsNotificationSnapshot(items) {
        try {
            window.localStorage.setItem(
                getResultsNotificationStorageKey(),
                JSON.stringify(buildResultsNotificationSnapshot(items))
            );
        } catch (error) {
            console.warn('SIPATUJU Residen - snapshot notifikasi tidak dapat disimpan:', error);
        }
    }

    function getUnreadResultsUpdateCount(items) {
        const currentSnapshot = buildResultsNotificationSnapshot(items);
        const seenSnapshot = readSeenResultsNotificationSnapshot();

        /*
         * Pertama kali fitur dipakai: jadikan kondisi saat ini sebagai baseline,
         * sehingga data lama tidak langsung dianggap sebagai notifikasi baru.
         */
        if (!seenSnapshot) {
            writeSeenResultsNotificationSnapshot(items);
            return 0;
        }

        let unreadCount = 0;

        Object.keys(currentSnapshot).forEach(function (key) {
            /*
             * Record baru tidak dihitung sebagai perubahan status.
             * Notifikasi hanya untuk pengajuan yang sebelumnya sudah dikenal,
             * lalu mengalami perubahan status/detail pemeriksaan.
             */
            if (!Object.prototype.hasOwnProperty.call(seenSnapshot, key)) {
                return;
            }

            if (seenSnapshot[key] !== currentSnapshot[key]) {
                unreadCount += 1;
            }
        });

        return unreadCount;
    }

    function renderResultsUpdateNotification(count) {
        const unreadCount = Math.max(0, Number(count) || 0);
        const hasUnread = unreadCount > 0;

        if (resultsNavAlert) {
            resultsNavAlert.hidden = !hasUnread;
        }

        if (resultsNavCount) {
            resultsNavCount.hidden = !hasUnread;
            resultsNavCount.textContent = hasUnread ? String(unreadCount) : '0';
        }
    }

    async function loadResultsNotification(){
        if(resultsNotificationRequestInFlight) return;

        const config = getResultsConfig();
        if(!config || !config.ajaxUrl || !config.nonce) return;

        resultsNotificationRequestInFlight = true;

        try{
            const body = new URLSearchParams();
            body.append('action', 'spj26_resident_results_data');
            body.append('nonce', config.nonce);

            const response = await fetch(config.ajaxUrl, {
                method:'POST',
                credentials:'same-origin',
                headers:{
                    'Content-Type':'application/x-www-form-urlencoded; charset=UTF-8'
                },
                body:body.toString()
            });

            const result = await response.json();
            if(!response.ok || !result || result.success !== true) return;

            const items = result.data && Array.isArray(result.data.submissions)
                ? result.data.submissions
                : [];

            renderResultsUpdateNotification(
                getUnreadResultsUpdateCount(items)
            );
        }
        catch(error){
            console.error('SIPATUJU Residen - notifikasi hasil:', error);
        }
        finally{
            resultsNotificationRequestInFlight = false;
        }
    }

    function startResultsNotificationRefresh(){
        if(resultsNotificationTimer){
            window.clearInterval(resultsNotificationTimer);
        }

        resultsNotificationTimer = window.setInterval(
            loadResultsNotification,
            30000
        );
    }

    function updateStepper(){

        const icons = {

            1:"article",
            2:"subject",
            3:"school",
            4:"library_books",
            5:"task_alt"

        };


        root
            .querySelectorAll(
                "[data-step-indicator]"
            )
            .forEach(
                function(item){


                    const step =
                        Number(
                            item.getAttribute(
                                "data-step-indicator"
                            )
                        );


                    const icon =
                        item.querySelector(
                            ".material-symbols-outlined"
                        );


                    item.classList.remove(
                        "is-active",
                        "is-complete"
                    );


                    if(step < currentStep){

                        item.classList.add(
                            "is-complete"
                        );


                        if(icon){
                            icon.textContent = "check";
                        }

                    }
                    else if(step === currentStep){

                        item.classList.add(
                            "is-active"
                        );


                        if(icon){
                            icon.textContent =
                                icons[step];
                        }

                    }
                    else{

                        if(icon){
                            icon.textContent =
                                icons[step];
                        }

                    }


                }
            );


        root
            .querySelectorAll(
                "[data-step-panel]"
            )
            .forEach(
                function(panel){


                    const step =
                        Number(
                            panel.getAttribute(
                                "data-step-panel"
                            )
                        );


                    panel.classList.toggle(
                        "is-active",
                        step === currentStep
                    );


                }
            );


        if(stepLine){

            const width =
                (
                    (
                        currentStep - 1
                    )
                    /
                    (
                        totalSteps - 1
                    )
                )
                *
                80;


            stepLine.style.width =
                width + "%";

        }


        clearError();


        const rightPane =
            root.querySelector(
                ".spj26res-rightpane"
            );


        if(
            window.innerWidth > 800
            &&
            rightPane
        ){

            rightPane.scrollTo({
                top:0,
                behavior:"smooth"
            });

        }
        else{

            window.scrollTo({
                top:0,
                behavior:"smooth"
            });

        }

    }

    function isValidHttpUrl(value){

        try{

            const url =
                new URL(
                    value
                );


            return (
                url.protocol === "https:"
                ||
                url.protocol === "http:"
            );

        }
        catch(error){

            return false;

        }

    }

       DUPLICATE JOURNAL
    ===================================================== */

    function normalizeJournalName(value){

        return String(
            value
            ||
            ""
        )
        .trim()
        .toLowerCase()
        .replace(
            /\s+/g,
            " "
        )
        .replace(
            /[.,;:()[\]{}'"`]/g,
            ""
        )
        .trim();

    }


    function normalizeIssn(value){

        return String(
            value
            ||
            ""
        )
        .toUpperCase()
        .replace(
            /[^0-9X]/g,
            ""
        );

    }


    /* =====================================================
       RESTRICTED JOURNAL CHECK

       Logic:
       - Nama jurnal cocok ATAU ISSN cocok => DITOLAK
       - Nama jurnal tidak cocok DAN ISSN tidak cocok => BOLEH

       Source of truth tetap backend SIPATUJU.
       Frontend tidak mengunduh seluruh daftar restricted.
    ===================================================== */

    function collectJournalRestrictionPayload(){

        const journals = [];

        for(
            let i = 1;
            i <= 3;
            i++
        ){

            journals.push({
                order:i,
                journal_name:
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-name"
                    ),
                issn:
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-issn"
                    )
            });

        }

        return journals;

    }


    async function checkRestrictedJournals(){

        const config =
            getSubmissionConfig();

        if(
            !config
            ||
            !config.ajaxUrl
            ||
            !config.nonce
        ){

            throw new Error(
                "Konfigurasi pemeriksaan jurnal belum tersedia."
            );

        }


        const body =
            new URLSearchParams();

        body.append(
            "action",
            "spj26_check_restricted_journals"
        );

        body.append(
            "nonce",
            config.nonce
        );

        body.append(
            "journals",
            JSON.stringify(
                collectJournalRestrictionPayload()
            )
        );


        const response =
            await fetch(
                config.ajaxUrl,
                {
                    method:"POST",
                    credentials:"same-origin",
                    headers:{
                        "Content-Type":
                            "application/x-www-form-urlencoded; charset=UTF-8"
                    },
                    body:
                        body.toString()
                }
            );


        let result = null;

        try{
            result =
                await response.json();
        }
        catch(error){
            throw new Error(
                "Respons pemeriksaan jurnal dari server tidak valid."
            );
        }


        if(
            !response.ok
            ||
            !result
            ||
            result.success !== true
        ){

            throw new Error(
                result
                &&
                result.data
                &&
                result.data.message
                ?
                result.data.message
                :
                "Pemeriksaan daftar Predator / Discontinued gagal dilakukan."
            );

        }


        return result.data || {};

    }


    function restrictedJournalMessage(matches){

        const list =
            Array.isArray(matches)
            ? matches
            : [];

        if(list.length === 0){
            return "";
        }


        return list
            .map(function(match){

                const order =
                    Number(match.order) || 0;

                const nameMatch =
                    match.name_match === true;

                const issnMatch =
                    match.issn_match === true;

                let reason =
                    "terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU";

                if(nameMatch && issnMatch){
                    reason =
                        "nama jurnal dan ISSN terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU";
                }
                else if(nameMatch){
                    reason =
                        "nama jurnal terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU";
                }
                else if(issnMatch){
                    reason =
                        "ISSN terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU";
                }


                return (
                    "Jurnal Kandidat "
                    +
                    order
                    +
                    " tidak dapat diajukan karena "
                    +
                    reason
                    +
                    "."
                );

            })
            .join(" ")
            +
            " Silakan ganti jurnal tersebut sebelum melanjutkan.";

    }


    async function validateRestrictedJournals(){

        const data =
            await checkRestrictedJournals();

        const matches =
            Array.isArray(data.matches)
            ? data.matches
            : [];

        const allowed =
            data.allowed === true
            &&
            matches.length === 0;

        if(!allowed){

            showError(
                restrictedJournalMessage(matches)
                ||
                "Satu atau lebih jurnal kandidat terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU."
            );

            return false;

        }

        return true;

    }


    function findDuplicateJournals(){

        const journals = [];


        for(
            let i = 1;
            i <= 3;
            i++
        ){

            journals.push({

                number:
                    i,

                name:
                    normalizeJournalName(
                        valueOf(
                            "#spj26res-submit-j"
                            +
                            i
                            +
                            "-name"
                        )
                    ),

                issn:
                    normalizeIssn(
                        valueOf(
                            "#spj26res-submit-j"
                            +
                            i
                            +
                            "-issn"
                        )
                    )

            });

        }


        for(
            let i = 0;
            i < journals.length;
            i++
        ){

            for(
                let j = i + 1;
                j < journals.length;
                j++
            ){

                const sameName =
                    journals[i].name !== ""
                    &&
                    journals[i].name
                    ===
                    journals[j].name;


                const sameIssn =
                    journals[i].issn !== ""
                    &&
                    journals[i].issn
                    ===
                    journals[j].issn;


                if(
                    sameName
                    ||
                    sameIssn
                ){

                    return {

                        first:
                            journals[i].number,

                        second:
                            journals[j].number,

                        reason:
                            sameName
                            &&
                            sameIssn
                            ?
                            "nama jurnal dan ISSN sama"
                            :
                            (
                                sameName
                                ?
                                "nama jurnal sama"
                                :
                                "ISSN sama"
                            )

                    };

                }

            }

        }


        return null;

    }


    /* =====================================================
       VALIDATION
    ===================================================== */

    function validateStep(step){

        clearError();


        /* =============================================
           STEP 1
        ============================================== */

        if(step === 1){

            if(
                !getSelectedArticleType()
            ){

                showError(
                    "Pilih jenis artikel terlebih dahulu."
                );


                return false;

            }

        }


        /* =============================================
           STEP 2
        ============================================== */

        if(step === 2){

            const thesisTitle =
                valueOf(
                    "#spj26res-submit-thesis-title"
                );


            const articleTitle =
                valueOf(
                    "#spj26res-submit-title"
                );


            const abstract =
                valueOf(
                    "#spj26res-submit-abstract"
                );


            if(!thesisTitle){

                showError(
                    "Judul Proposal Tesis wajib diisi."
                );


                return false;

            }


            if(!articleTitle){

                showError(
                    "Judul Artikel wajib diisi."
                );


                return false;

            }


            if(!abstract){

                showError(
                    "Abstrak artikel wajib diisi."
                );


                return false;

            }

        }


        /* =============================================
           STEP 3
        ============================================== */

        if(step === 3){

            const supervisor =
                valueOf(
                    "#spj26res-submit-supervisor"
                );


            if(!supervisor){

                showError(
                    "Dosen Pembimbing Utama wajib diisi."
                );


                return false;

            }

        }


        /* =============================================
           STEP 4
        ============================================== */

        if(step === 4){

            for(
                let i = 1;
                i <= 3;
                i++
            ){

                const name =
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-name"
                    );


                const issn =
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-issn"
                    );


                const quartile =
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-index"
                    );


                const link =
                    valueOf(
                        "#spj26res-submit-j"
                        +
                        i
                        +
                        "-link"
                    );


                if(
                    !name
                    ||
                    !issn
                    ||
                    !quartile
                    ||
                    !link
                ){

                    showError(
                        "Lengkapi Nama Jurnal, ISSN, Kuartil Jurnal, dan Link Jurnal pada ketiga jurnal kandidat."
                    );


                    return false;

                }


                if(
                    !isValidHttpUrl(
                        link
                    )
                ){

                    showError(
                        "Link Jurnal Kandidat "
                        +
                        i
                        +
                        " tidak valid. Gunakan alamat lengkap yang diawali https:// atau http://."
                    );


                    return false;

                }

            }


            const duplicate =
                findDuplicateJournals();


            if(duplicate){

                showError(
                    "Jurnal Kandidat "
                    +
                    duplicate.first
                    +
                    " dan Jurnal Kandidat "
                    +
                    duplicate.second
                    +
                    " terdeteksi sama karena "
                    +
                    duplicate.reason
                    +
                    ". Gunakan tiga jurnal yang berbeda."
                );


                return false;

            }

        }


        return true;

    }

       NEXT BUTTONS

       STEP 4 -> STEP 5 menjalankan restricted-journal precheck
       sebelum user dapat masuk ke Finalisasi.
    ===================================================== */

    root
        .querySelectorAll(
            "[data-next-step]"
        )
        .forEach(
            function(button){


                button.addEventListener(
                    "click",
                    async function(){


                        if(
                            !validateStep(
                                currentStep
                            )
                        ){

                            return;

                        }


                        const next =
                            Number(
                                button.getAttribute(
                                    "data-next-step"
                                )
                            );


                        if(
                            currentStep === 4
                            &&
                            next === 5
                        ){

                            const originalHtml =
                                button.innerHTML;

                            button.disabled =
                                true;

                            button.innerHTML =
                                '<span>Memeriksa...</span>'
                                +
                                '<span class="material-symbols-outlined">hourglass_top</span>';

                            try{

                                const allowed =
                                    await validateRestrictedJournals();

                                if(!allowed){
                                    return;
                                }

                            }
                            catch(error){

                                showError(
                                    error
                                    &&
                                    error.message
                                    ?
                                    error.message
                                    :
                                    "Pemeriksaan daftar Predator / Discontinued gagal dilakukan."
                                );

                                return;

                            }
                            finally{

                                button.disabled =
                                    false;

                                button.innerHTML =
                                    originalHtml;

                            }

                        }


                        if(next === 5){

                            buildSummary();

                        }


                        currentStep =
                            next;


                        updateStepper();


                    }
                );


            }
        );

        );


    /* =====================================================
       ABSTRACT WORD COUNTER
    ===================================================== */

    const abstractTextarea =
        root.querySelector(
            "#spj26res-submit-abstract"
        );


    const wordCounter =
        root.querySelector(
            "#spj26res-submit-word-counter"
        );


    function updateWordCounter(){

        if(
            !abstractTextarea
            ||
            !wordCounter
        ){

            return;

        }


        const text =
            abstractTextarea.value.trim();


        const count =
            text === ""
            ?
            0
            :
            text
                .split(/\s+/)
                .filter(Boolean)
                .length;


        wordCounter.textContent =
            count
            +
            " kata";

    }


    if(abstractTextarea){

        abstractTextarea.addEventListener(
            "input",
            updateWordCounter
        );


        updateWordCounter();

    }


    /* =====================================================
       SUCCESS
    ===================================================== */

    const successBox =
        root.querySelector(
            "#spj26res-submit-success"
        );


    const successMessage =
        root.querySelector(
            "#spj26res-submit-success-message"
        );


    const successCode =
        root.querySelector(
            "#spj26res-submit-success-code"
        );


    function clearSuccess(){

        if(successBox){

            successBox.classList.remove(
                "is-visible"
            );

        }


        if(successMessage){

            successMessage.textContent =
                "";

        }


        if(successCode){

            successCode.textContent =
                "";

        }

    }


    function showSuccess(
        message,
        code,
        status
    ){

        clearError();


        if(successMessage){

            successMessage.textContent =
                message
                ||
                "Pengajuan berhasil dikirim.";

        }


        if(successCode){

            successCode.textContent =
                "Kode Pengajuan: "
                +
                (
                    code
                    ||
                    "-"
                )
                +
                (
                    status
                    ?
                    " • "
                    +
                    status
                    :
                    ""
                );

        }


        if(successBox){

            successBox.classList.add(
                "is-visible"
            );


            successBox.scrollIntoView({
                behavior:"smooth",
                block:"center"
            });

        }

    }


    /* =====================================================
       BACKEND SUBMISSION

       UI:
       Original Article
       Systematic Review / Meta-Analysis

       BACKEND:
       Research Article
       Review Article
    ===================================================== */

    async function submitToBackend(){

        const config =
            getSubmissionConfig();


        if(
            !config
            ||
            !config.ajaxUrl
            ||
            !config.nonce
            ||
            !config.submissionToken
        ){

            throw new Error(
                "Konfigurasi Submission System belum tersedia."
            );

        }


        const articleType =
            getSelectedArticleType();


        if(!articleType){

            throw new Error(
                "Jenis artikel belum dipilih."
            );

        }


        const body =
            new URLSearchParams();


        body.append(
            "action",
            "spj26_create_submission"
        );


        body.append(
            "nonce",
            config.nonce
        );


        body.append(
            "request_token",
            config.submissionToken
        );


        /*
         * Backend compatible value.
         */

        body.append(
            "article_type",
            articleType.value
        );


        /*
         * Human-friendly label.
         * Older backend may ignore it safely.
         */

        body.append(
            "article_type_display",
            articleType.dataset.displayLabel
            ||
            ""
        );


        body.append(
            "thesis_proposal_title",
            valueOf(
                "#spj26res-submit-thesis-title"
            )
        );


        body.append(
            "article_title",
            valueOf(
                "#spj26res-submit-title"
            )
        );


        body.append(
            "article_abstract",
            valueOf(
                "#spj26res-submit-abstract"
            )
        );


        body.append(
            "supervisor_name",
            valueOf(
                "#spj26res-submit-supervisor"
            )
        );


        for(
            let i = 1;
            i <= 3;
            i++
        ){

            body.append(
                "journal_"
                +
                i
                +
                "_name",
                valueOf(
                    "#spj26res-submit-j"
                    +
                    i
                    +
                    "-name"
                )
            );


            body.append(
                "journal_"
                +
                i
                +
                "_issn",
                valueOf(
                    "#spj26res-submit-j"
                    +
                    i
                    +
                    "-issn"
                )
            );


            body.append(
                "journal_"
                +
                i
                +
                "_quartile",
                valueOf(
                    "#spj26res-submit-j"
                    +
                    i
                    +
                    "-index"
                )
            );


            body.append(
                "journal_"
                +
                i
                +
                "_link",
                valueOf(
                    "#spj26res-submit-j"
                    +
                    i
                    +
                    "-link"
                )
            );

        }


        const response =
            await fetch(
                config.ajaxUrl,
                {

                    method:
                        "POST",

                    credentials:
                        "same-origin",

                    headers:{
                        "Content-Type":
                            "application/x-www-form-urlencoded; charset=UTF-8"
                    },

                    body:
                        body.toString()

                }
            );


        let result;


        try{

            result =
                await response.json();

        }
        catch(error){

            throw new Error(
                "Respons server tidak valid."
            );

        }


        if(
            !response.ok
            ||
            !result
            ||
            result.success !== true
        ){

            const message =
                result
                &&
                result.data
                &&
                result.data.message
                ?
                result.data.message
                :
                "Pengajuan gagal dikirim.";


            throw new Error(
                message
            );

        }


        return result.data
        ||
        {};

    }


    /* =====================================================
       FINAL SUBMIT
    ===================================================== */

    const finalButton =
        root.querySelector(
            "#spj26res-submit-final"
        );


    if(finalButton){

        finalButton.addEventListener(
            "click",
            async function(){


                clearError();

                clearSuccess();


                /*
                 * Revalidate all previous steps.
                 */

                for(
                    let step = 1;
                    step <= 4;
                    step++
                ){

                    if(
                        !validateStep(
                            step
                        )
                    ){

                        currentStep =
                            step;


                        updateStepper();


                        return;

                    }

                }


                /*
                 * Re-check restricted list immediately before final submit.
                 * Ini menutup race condition jika daftar restricted berubah
                 * setelah user meninggalkan Step 4.
                 */

                try{

                    const restrictionData =
                        await checkRestrictedJournals();

                    const restrictionMatches =
                        Array.isArray(
                            restrictionData.matches
                        )
                        ?
                        restrictionData.matches
                        :
                        [];

                    if(
                        restrictionData.allowed !== true
                        ||
                        restrictionMatches.length > 0
                    ){

                        currentStep = 4;

                        updateStepper();

                        showError(
                            restrictedJournalMessage(
                                restrictionMatches
                            )
                            ||
                            "Satu atau lebih jurnal kandidat terdapat dalam daftar Jurnal Predator / Discontinued SIPATUJU."
                        );

                        return;

                    }

                }
                catch(error){

                    currentStep = 4;

                    updateStepper();

                    showError(
                        error
                        &&
                        error.message
                        ?
                        error.message
                        :
                        "Pemeriksaan daftar Predator / Discontinued gagal dilakukan."
                    );

                    return;

                }


                buildSummary();


                const originalHtml =
                    finalButton.innerHTML;


                finalButton.disabled =
                    true;


                finalButton.innerHTML =
                    '<span>Mengirim...</span>'
                    +
                    '<span class="material-symbols-outlined">hourglass_top</span>';


                try{


                    const data =
                        await submitToBackend();


                    showSuccess(

                        data.message
                        ||
                        "Pengajuan berhasil dikirim.",

                        data.code
                        ||
                        "",

                        data.status_label
                        ||
                        "Menunggu Pemeriksaan"

                    );


                    finalButton.innerHTML =
                        '<span>Pengajuan Terkirim</span>'
                        +
                        '<span class="material-symbols-outlined">check_circle</span>';


                    finalButton.disabled =
                        true;


                    root
                        .querySelectorAll(
                            "input, textarea, select"
                        )
                        .forEach(
                            function(field){

                                field.disabled =
                                    true;

                            }
                        );


                    const resultsUrl =
                        (
                            window.spj26ResidentResultsConfig
                            &&
                            window.spj26ResidentResultsConfig.resultsUrl
                        )
                        ||
                        "/residen-sipatuju/hasil/";


                    const submittedCode =
                        data.code
                        ||
                        "";


                    window.setTimeout(
                        function(){

                            const target =
                                new URL(
                                    resultsUrl,
                                    window.location.origin
                                );


                            if(submittedCode){

                                target.searchParams.set(
                                    "submitted",
                                    submittedCode
                                );

                            }


                            window.location.assign(
                                target.toString()
                            );

                        },
                        1200
                    );


                }
                catch(error){


                    finalButton.disabled =
                        false;


                    finalButton.innerHTML =
                        originalHtml;


                    showError(
                        error
                        &&
                        error.message
                        ?
                        error.message
                        :
                        "Pengajuan gagal dikirim."
                    );


                }


            }
        );

    }
    /* =====================================================
       INITIALIZATION
    ===================================================== */

    function initialize(){

        applyResidentIdentity();

        applyNavigationConfig();

        loadResultsNotification();

        startResultsNotificationRefresh();

        updateStepper();

    }


    if(
        document.readyState === "loading"
    ){

        document.addEventListener(
            "DOMContentLoaded",
            initialize,
            {
                once:true
            }
        );

    }
    else{

        initialize();

    }


})();

</script>
